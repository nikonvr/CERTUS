@functools.lru_cache(maxsize=None)
def n_of(mat, wl):
    if mat == SUB:
        return complex(get_n_substrate_array_by_id(0, np.array([float(wl)]))[0])
    return complex(db._db.get_refractive_index(mat, float(wl)))


# ---- operations from the design and the plan ------------------------------------------------------------------------------
N = design["n_layers"]
assert N == 90
assert max(b["end"] for b in plan["blocks"]) == N
mats = [design["materials"][i] for i in design["material_indices"]]
thick = np.array(design["thickness_nm"], dtype=float)
rate = set(plan["rate_layers"])
assert not rate & set(range(9)), "no Rate before layer 10 (owner)"
lam_plan = [next(float(b["wavelength"]) for b in plan["blocks"] if b["start"] <= k < b["end"]) for k in range(N)]
ops = [dict(component_layer=k + 1, role="couche pièce", mode="Rate" if k in rate else "Optique") for k in range(N)]
assert all(m == (H if k % 2 == 0 else L) for k, m in enumerate(mats))
lam_mon = []
for k in range(N):
    if ops[k]["mode"] == "Optique":
        lam_mon.append(lam_plan[k])
    else:
        lam_mon.append(next(lam_plan[j] for j in range(k - 1, -1, -1) if ops[j]["mode"] == "Optique"))


def witness_T(th, wl):
    """T (fraction) of the witness for a stack th (substrate first) at one wavelength, back face included."""
    th = np.asarray(th, dtype=np.float64)
    nl = np.array([[n_of(mats[k], wl) for k in range(len(th))]], dtype=np.complex128)
    _, T, _ = calculate_RTRback_incoherent_vectorized(th, nl, np.array([n_of(SUB, wl)], dtype=np.complex128),
                                                      np.array([float(wl)]))
    return float(T[0])


# ---- per operation: QWOT, signals, turning points ------------------------------------------------------------------------
rows, curves, full = [], [], []
for k, o in enumerate(ops):
    wl, d = lam_mon[k], thick[k]
    n600, nmon, n550 = n_of(mats[k], QW), n_of(mats[k], wl), n_of(mats[k], 550.0)
    below = list(thick[:k])
    step = 0.1
    grid = np.append(np.arange(0.0, d, step), d)
    T = np.array([witness_T(below + [x], wl) for x in grid])
    dT = np.diff(T)
    ext = [i for i in range(1, len(dT)) if dT[i - 1] * dT[i] < 0]  # extrema crossed during the layer
    slope = (T[-1] - T[-2]) / (grid[-1] - grid[-2])
    rows.append(dict(op=k + 1, comp=o["component_layer"], role=o["role"], mat=mats[k], d=d, n600=n600.real, k600=-n600.imag,
                     qwot=4 * n600.real * d / QW, n550=n550.real, ot550=n550.real * d, mode=o["mode"], wl=wl, nmon=nmon.real,
                     T0=100 * T[0], T1=100 * T[-1], ntp=len(ext), tp=[(round(float(grid[i]), 1), round(100 * float(T[i]), 3)) for i in ext],
                     slope=100 * slope, sens="montant" if slope > 0 else "descendant", nsub=n_of(SUB, wl).real))
    sub = np.unique(np.append(np.linspace(0, len(grid) - 1, 41).round().astype(int), np.array(ext, dtype=int)))
    curves.append([(float(grid[i]), 100 * float(T[i])) for i in sub])
    full.append((grid.copy(), 100 * T, [int(i) for i in ext]))

rmse = float(np.sqrt(np.mean((100 * Tpart - target[:, 1]) ** 2)))

def stack_T(th, ms, w_arr):
    nl = np.array([[n_of(m, float(w)) for m in ms] for w in w_arr], dtype=np.complex128)
    _, T, _ = calculate_RTRback_incoherent_vectorized(np.asarray(th, dtype=np.float64), nl,
                                                       np.array([n_of(SUB, float(w)) for w in w_arr], dtype=np.complex128), w_arr.astype(np.float64))
    return 100 * T
