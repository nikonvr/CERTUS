"""Replay the fixed historical 99-layer partition through existing production tools."""

import argparse
import hashlib
import io
import json
import math
import os
import re
import shutil
import subprocess
import sys
import time
from contextlib import redirect_stdout
from datetime import datetime, timezone
from pathlib import Path

ROOT = next(parent for parent in Path(__file__).resolve().parents if (parent / "scripts/bench_examples.py").is_file())
PARTS = ((0, 22), (22, 42), (42, 76), (76, 99))
SEEDS = (42, 101, 202)
FULL = "example/example_strat/JSON-strat-bandpass-5cav-99c.json"
DEADLINE = datetime.fromisoformat("2026-10-21T08:00:00+00:00")


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def git(*args):
    return subprocess.check_output(["git", *args], cwd=ROOT, text=True).strip()


def write(path, payload):
    temporary = path.with_suffix(path.suffix + ".tmp")
    temporary.write_text(json.dumps(payload, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
    os.replace(temporary, path)


def child(args, slot):
    os.chdir(ROOT)
    os.environ["QT_QPA_PLATFORM"] = "offscreen"
    os.environ["CERTUS_BENCH_TIMEOUT_S"] = "21600"
    sys.path.insert(0, str(ROOT))
    sys.path.insert(0, str(ROOT / "scripts"))
    import bench_examples as bench

    bench.qapp()
    bench.autoanswer_dialogs(True)
    import numpy as np
    import campagne_intervalles as campaign

    assert len(json.loads((ROOT / FULL).read_text(encoding="utf-8"))["stack_multipliers"]) == 99
    if args.child == "interval":
        from CERTUS_STRAT import CertusStratApp

        audits = []
        returned = []
        original_workflow = CertusStratApp.run_workflow
        original_wait = bench.wait_for

        def workflow(app, *positional, **keywords):
            collect = app.collect_params

            def audited_collect(*positional, **keywords):
                params = collect(*positional, **keywords)
                count = len([v for v in params["stack_string"].split(",") if v.strip()])
                assert count == args.b - args.a
                assert params["execution_mode"] == "fast"
                assert params["robustness_seed"] == args.seed
                assert params["noise_layer_offset"] == args.a
                assert params["noise_total_layers"] == 99
                assert params["monochromator_resolution_nm"] == 2.0
                assert params["search_resolution"] is False
                assert params["index_corridor"] == 0.005
                assert params["strat_local_search"] is True
                assert params["tp_hysteresis_floor"] == 0.000415
                assert not params.get("injected_strategies")
                keys = ("execution_mode", "robustness_seed", "phase_a_seed", "noise_layer_offset", "noise_total_layers",
                        "monochromator_resolution_nm", "search_resolution", "index_corridor", "strat_local_search",
                        "tp_hysteresis_floor", "stack_string", "nH_id", "nL_id", "nSub_id", "l0", "wl_range", "wl_step")
                audits.append({key: params.get(key) for key in keys})
                return params

            app.collect_params = audited_collect
            return original_workflow(app, *positional, **keywords)

        def observed_wait(*positional, **keywords):
            result = original_wait(*positional, **keywords)
            returned.append(result)
            return result

        CertusStratApp.run_workflow = workflow
        bench.wait_for = observed_wait
        row = campaign.mesurer(args.a, args.b, mode="fast", cache_dir=slot, seed=args.seed)
        write(slot / "params_effectifs.json", audits)
        write(campaign.fichier(args.a, args.b, slot, args.seed), row)
        measured = row["verdict"] in {"DEPOSABLE", "AUCUNE_DEPOSABLE"} and len(audits) >= 2
        valid = measured and row["verdict"] == "DEPOSABLE"
        if valid:
            assert math.isfinite(row["score"]) and row["score"] > 0
            suffix = campaign._suffixe_graine(args.seed)
            thickness_path = slot / f"th_{args.a:03d}_{args.b:03d}{suffix}.npy"
            thicknesses = np.load(thickness_path)
            assert thicknesses.ndim == 2 and thicknesses.shape[1] == args.b - args.a and thicknesses.shape[0] > 0
            row.update(thickness_sha256=sha(thickness_path), thickness_shape=list(thicknesses.shape))
        if returned and isinstance(returned[-1], dict):
            results = (returned[-1].get("final_results") or {}).get("all_strategies_results", [])
            reduced = [{key: result.get(key) for key in ("strategy", "robustness_score", "crash_rate", "crash_eliminated")}
                       for result in results]
            with (slot / "returned_plans_and_scores.json").open("x", encoding="utf-8") as output:
                json.dump(reduced, output, default=lambda value: value.tolist(), ensure_ascii=False)
        write(slot / "result.json", {"measured": measured, "valid": valid, "row": row, "collect_calls": len(audits)})
        return 0 if measured else 1

    import classer_partitions as assembly

    __import__("CERTUS_STRAT")  # Complete console imports before redirect_stdout.

    cache = slot / "cache"
    assert cache.is_dir()
    loaded = assembly.charger(cache, args.seed)
    assert set(loaded) == set(PARTS)
    assert len({loaded[part]["th"].shape[0] for part in PARTS}) == 1
    assert sum(loaded[part]["th"].shape[1] for part in PARTS) == 99
    sys.argv = ["classer_partitions.py", "--cache-dir", str(cache), "--mode", "fast", "--graine", str(args.seed)]
    output = io.StringIO()
    with redirect_stdout(output):
        exit_code = assembly.main()
    printed = output.getvalue()
    print(printed, end="", flush=True)
    matches = re.findall(r"^\s*\d+\s+4\s+0-22\s+22-42\s+42-76\s+76-99\s+([0-9.]+)\s", printed, re.MULTILINE)
    assert len(matches) <= 1
    valid = exit_code == 0 and len(matches) == 1
    write(slot / "result.json", {"measured": exit_code == 0, "valid": valid,
          "seel_nm_printed": float(matches[0]) if valid else None, "printed_resolution_nm": 0.001,
          "method": "Unchanged classer_partitions.main, P95 spectrum of concatenated actual thickness draws",
          "reason": "healthy_fixed_partition" if valid else "no_complete_healthy_partition"})
    return exit_code


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--outdir", required=True)
    parser.add_argument("--prepare", action="store_true")
    parser.add_argument("--resume-interrupted", action="store_true")
    parser.add_argument("--child", choices=("interval", "assembly"))
    parser.add_argument("--slot")
    parser.add_argument("--seed", type=int)
    parser.add_argument("--a", type=int)
    parser.add_argument("--b", type=int)
    args = parser.parse_args()
    folder = (ROOT / args.outdir).resolve()
    assert folder.is_relative_to(ROOT / "reports")
    if args.child:
        slot = (ROOT / args.slot).resolve()
        assert slot.is_relative_to(folder)
        return child(args, slot)
    frozen = folder / "run_A3_retained_intervals.py.txt"
    protocol = folder / "intervalles_protocole.json"
    if args.prepare:
        folder.mkdir(parents=True, exist_ok=True)
        assert not frozen.exists() and not protocol.exists()
        frozen.write_bytes(Path(__file__).read_bytes())
        write(protocol, {"prepared_utc": datetime.now(timezone.utc).isoformat(), "source": FULL,
              "source_sha256": sha(ROOT / FULL), "parts": PARTS, "seeds": SEEDS, "mode": "fast",
              "campaign_tool_sha256": sha(ROOT / "scripts/campagne_intervalles.py"),
              "assembly_tool_sha256": sha(ROOT / "scripts/classer_partitions.py"),
              "slit_nm": 2.0, "search_resolution": False, "deadline_utc": DEADLINE.isoformat(),
              "independent_confirmation": False, "scope": "Three searches of the same previously selected four-witness partition; no new partition optimization",
              "assembly": "Existing classer_partitions, all four intervals share one seed and preserve physical read-noise offsets",
              "script_sha256": sha(frozen)})
        print("A3_INTERVALS=PREPARED seeds=42,101,202 parts=0-22/22-42/42-76/76-99", flush=True)
        return 0
    assert frozen.read_bytes() == Path(__file__).read_bytes()
    assert not git("status", "--porcelain", "--untracked-files=no")
    head = git("rev-parse", "HEAD")
    fixed = json.loads(protocol.read_text(encoding="utf-8"))
    assert sha(ROOT / FULL) == fixed["source_sha256"]
    assert sha(ROOT / "scripts/campagne_intervalles.py") == fixed["campaign_tool_sha256"]
    assert sha(ROOT / "scripts/classer_partitions.py") == fixed["assembly_tool_sha256"]
    state_path = folder / "intervalles_etat.json"
    if state_path.exists():
        state = json.loads(state_path.read_text(encoding="utf-8"))
        assert state["head"] == head
        assert args.resume_interrupted or state["status"] == "complete", "Inspect processes before resuming"
    else:
        state = {"head": head, "attempts": [], "status": "prepared"}
    state["controller_pid"] = os.getpid()
    for seed in SEEDS:
        jobs = [("interval", a, b) for a, b in PARTS] + [("assembly", None, None)]
        for kind, a, b in jobs:
            key = f"s{seed:03d}_{kind}" + (f"_{a:03d}_{b:03d}" if a is not None else "")
            if any(old["key"] == key and old.get("measured") for old in state["attempts"]):
                continue
            if datetime.now(timezone.utc) >= DEADLINE:
                state["status"] = "deadline_reached"
                write(state_path, state)
                return 0
            assert git("rev-parse", "HEAD") == head and not git("status", "--porcelain", "--untracked-files=no")
            chosen = []
            if kind == "assembly":
                for x, y in PARTS:
                    matches = [old for old in state["attempts"] if old["key"] == f"s{seed:03d}_interval_{x:03d}_{y:03d}" and old.get("valid")]
                    if matches:
                        chosen.append(matches[-1])
                if len(chosen) != 4:
                    state.setdefault("unassembled_seeds", []).append(seed)
                    write(state_path, state)
                    continue
            attempt = 1 + sum(old["key"] == key for old in state["attempts"])
            slot = folder / "intervalles" / f"{key}_a{attempt:02d}"
            slot.mkdir(parents=True, exist_ok=False)
            if kind == "assembly":
                cache = slot / "cache"
                cache.mkdir()
                for selected in chosen:
                    source_slot = ROOT / selected["slot"]
                    for path in source_slot.iterdir():
                        if path.name.startswith(("i_", "th_", "cfg_")):
                            shutil.copy2(path, cache / path.name)
            record = {"key": key, "attempt": attempt, "slot": slot.relative_to(ROOT).as_posix(), "seed": seed,
                      "started_utc": datetime.now(timezone.utc).isoformat(), "measured": False, "valid": False}
            state["attempts"].append(record)
            state.update(status="running", current_key=key)
            write(state_path, state)
            command = [sys.executable, str(frozen), "--outdir", args.outdir, "--child", kind,
                       "--slot", record["slot"], "--seed", str(seed)]
            if a is not None:
                command.extend(["--a", str(a), "--b", str(b)])
            environment = dict(os.environ, QT_QPA_PLATFORM="offscreen", PYTHONIOENCODING="utf-8", CERTUS_BENCH_TIMEOUT_S="21600")
            for name in ("CERTUS_PROBE_OVERRIDES", "CERTUS_PROBE_TAG", "CERTUS_DESIGN_JSON"):
                environment.pop(name, None)
            print("START", key, flush=True)
            started = time.monotonic()
            with (slot / "raw.log").open("xb") as output:
                process = subprocess.Popen(command, cwd=ROOT, env=environment, stdout=output, stderr=subprocess.STDOUT,
                                           creationflags=subprocess.CREATE_NO_WINDOW if os.name == "nt" else 0)
                record["child_pid"] = process.pid
                write(state_path, state)
                code = process.wait()
            result_path = slot / "result.json"
            result = json.loads(result_path.read_text(encoding="utf-8")) if result_path.exists() else {}
            record.update(exit_code=code, wall_s=time.monotonic() - started, finished_utc=datetime.now(timezone.utc).isoformat(),
                          result=result, measured=code == 0 and bool(result.get("measured")), valid=code == 0 and bool(result.get("valid")),
                          raw_sha256=sha(slot / "raw.log"))
            state["status"] = "case_complete" if record["measured"] else "failed"
            write(state_path, state)
            print("END", key, "measured=" + str(record["measured"]), "valid=" + str(record["valid"]), flush=True)
            if not record["measured"]:
                return 1
    state["status"] = "complete"
    write(state_path, state)
    return 0


if __name__ == "__main__":
    code = main()
    sys.stdout.flush()
    sys.stderr.flush()
    os._exit(code)
