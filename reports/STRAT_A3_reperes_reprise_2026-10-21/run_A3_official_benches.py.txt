"""Sequential official A3 benches; preserve attempts and reject missing anchors."""

import argparse
import hashlib
import json
import math
import os
import re
import subprocess
import sys
import time
from datetime import datetime, timezone
from pathlib import Path

ROOT = next(parent for parent in Path(__file__).resolve().parents if (parent / "scripts/bench_examples.py").is_file())
sys.path.insert(0, str(ROOT))
from scripts.resumer_artefact_sonde import seel

EXAMPLES = Path("example/example_strat")
CASES = (
    ("48c_premium", EXAMPLES / "JSON-strat-example.json", 48, "premium"),
    ("75c_fast", EXAMPLES / "JSON-strat-random75.json", 75, "fast"),
    ("35c_deep", None, 35, "deep"),
    ("3cav_livre_36c_deep", EXAMPLES / "JSON-strat-bandpass-3cav-deep.json", 36, "deep"),
    ("r75x2_native_1nm_deep", EXAMPLES / "JSON-strat-random75-x2-fabricable.json", 75, "deep"),
    ("r75x2_rampes_2nm_deep", EXAMPLES / "JSON-strat-random75-x2-fabricable-2nm.json", 75, "deep"),
)


def git(*args):
    return subprocess.check_output(["git", *args], cwd=ROOT).decode("utf-8").strip()


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def write_json(path, payload):
    temporary = path.with_suffix(path.suffix + ".tmp")
    temporary.write_text(json.dumps(payload, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    os.replace(temporary, path)


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--outdir", default="reports/STRAT_A3_reperes_2026-10-21")
    parser.add_argument("--prepare", action="store_true")
    parser.add_argument("--resume-interrupted", action="store_true")
    parser.add_argument("--deadline-utc", default="2026-10-21T08:00:00+00:00")
    args = parser.parse_args()
    folder = (ROOT / args.outdir).resolve()
    if not folder.is_relative_to(ROOT / "reports"):
        raise ValueError("The output directory must be inside this worktree's reports")
    folder.mkdir(parents=True, exist_ok=True)
    generated = folder / "config_35c_deep_A3.json"
    frozen_program = folder / "run_A3_official_benches.py.txt"
    if args.prepare:
        baseline = json.loads((ROOT / EXAMPLES / "JSON-strat-bandpass-3cav.json").read_text(encoding="utf-8"))
        assert len(baseline["stack_multipliers"]) == 35
        derived = dict(baseline, execution_mode="deep")
        if generated.exists():
            assert json.loads(generated.read_text(encoding="utf-8")) == derived
        else:
            write_json(generated, derived)
        if frozen_program.exists():
            assert frozen_program.read_bytes() == Path(__file__).read_bytes()
        else:
            frozen_program.write_bytes(Path(__file__).read_bytes())

    plan = []
    for label, relative, count, mode in CASES:
        config_path = ROOT / relative if relative else generated
        config = json.loads(config_path.read_text(encoding="utf-8"))
        assert len(config["stack_multipliers"]) == count, label
        assert config["execution_mode"] == mode, label
        row = {
            "case": label, "config": config_path.relative_to(ROOT).as_posix(),
            "config_sha256": sha(config_path), "declared_layers": count,
            "configured_mode": mode, "robustness_seed": int(config.get("robustness_seed", 42) or 42),
            "slit_nm": float(config["monochromator_resolution_nm"]),
            "index_corridor": float(config["index_corridor"]),
        }
        if config.get("injected_strategies"):
            injection = ROOT / config["injected_strategies"]
            row.update(injection=injection.relative_to(ROOT).as_posix(), injection_sha256=sha(injection))
        plan.append(row)
    if args.prepare:
        for row in plan:
            print(f"PREPARED {row['case']} layers={row['declared_layers']} mode={row['configured_mode']} seed={row['robustness_seed']} slit={row['slit_nm']}")
        return 0

    if git("status", "--porcelain", "--untracked-files=no"):
        raise RuntimeError("Tracked changes must be committed before measuring")
    assert frozen_program.read_bytes() == Path(__file__).read_bytes(), "The frozen driver differs"
    head = git("rev-parse", "HEAD")
    deadline = datetime.fromisoformat(args.deadline_utc)
    state_path = folder / "etat.json"
    if state_path.exists():
        state = json.loads(state_path.read_text(encoding="utf-8"))
        assert state["head"] == head and state["plan"] == plan, "Source or inputs changed"
        if state["status"] != "complete" and not args.resume_interrupted:
            raise RuntimeError("Existing run: check all processes before --resume-interrupted")
    else:
        state = {"head": head, "driver_sha256": sha(Path(__file__)), "plan": plan, "attempts": [], "status": "prepared"}
    state["controller_pid"] = os.getpid()
    for row in plan:
        if any(old["case"] == row["case"] and old.get("valid") for old in state["attempts"]):
            continue
        if datetime.now(timezone.utc) >= deadline:
            state["status"] = "deadline_reached"
            write_json(state_path, state)
            return 0
        assert git("rev-parse", "HEAD") == head
        assert not git("status", "--porcelain", "--untracked-files=no")
        assert sha(ROOT / row["config"]) == row["config_sha256"]
        if row.get("injection"):
            assert sha(ROOT / row["injection"]) == row["injection_sha256"]
        attempt = 1 + sum(old["case"] == row["case"] for old in state["attempts"])
        stem = f"STRAT_bench_{row['case']}_{head[:8]}_defauts_2026-10-21_a{attempt:02d}"
        official = folder / (stem + ".txt")
        raw = folder / (stem + ".raw.log")
        if official.exists() or raw.exists():
            raise RuntimeError("Attempt files already exist; inspect before allocating a new slot")
        command = [sys.executable, "scripts/bench_examples.py", "strat", "--auto-yes", "--out", official.relative_to(ROOT).as_posix()]
        environment = dict(os.environ, CERTUS_DESIGN_JSON=row["config"], CERTUS_BENCH_TIMEOUT_S="21600", QT_QPA_PLATFORM="offscreen", PYTHONIOENCODING="utf-8")
        for key in ("CERTUS_PROBE_OVERRIDES", "CERTUS_PROBE_TAG"):
            environment.pop(key, None)
        record = {"case": row["case"], "attempt": attempt, "official": official.relative_to(ROOT).as_posix(), "raw": raw.relative_to(ROOT).as_posix(), "started_utc": datetime.now(timezone.utc).isoformat(), "valid": False}
        state["attempts"].append(record)
        state.update(status="running", current_case=row["case"])
        write_json(state_path, state)
        print("START", row["case"], flush=True)
        started = time.monotonic()
        with raw.open("xb") as output:
            child = subprocess.Popen(command, cwd=ROOT, env=environment, stdout=output, stderr=subprocess.STDOUT, creationflags=subprocess.CREATE_NO_WINDOW if os.name == "nt" else 0)
            record["child_pid"] = child.pid
            write_json(state_path, state)
            exit_code = child.wait()
        text = official.read_text(encoding="utf-8", errors="replace") if official.exists() else ""
        observed = dict(re.findall(r"^([A-Z0-9_]+)=(.*)$", text, flags=re.MULTILINE))
        observed.update(re.findall(r"^(STRAT_RANK[0-9]{2}) (.*)$", text, flags=re.MULTILINE))
        parse_error = None
        try:
            result = float(observed.get("RESULT", "nan"))
            eliminated = int(observed.get("STRAT_CRASH_ELIMINATED", "missing").split("/")[0])
            population = int(observed.get("STRAT_N_STRATEGIES", "0"))
        except ValueError as error:
            result, eliminated, population = math.nan, -1, 0
            parse_error = str(error)
        expected_ranks = all(f"STRAT_RANK{rank:02d}" in observed for rank in range(min(10, population)))
        valid = exit_code == 0 and math.isfinite(result) and result > 0 and population > 0 and eliminated == 0 and "WAIT_TIMEOUT" not in observed and expected_ranks
        valid = valid and observed.get("DESIGN_JSON") == Path(row["config"]).name
        record.update(exit_code=exit_code, wall_s=time.monotonic() - started, finished_utc=datetime.now(timezone.utc).isoformat(), observed=observed, valid=valid, parse_error=parse_error, seel_nm=seel(result) if valid else None, raw_sha256=sha(raw))
        if official.exists():
            record["official_sha256"] = sha(official)
        state["status"] = "case_complete" if valid else "failed"
        write_json(state_path, state)
        print("END", row["case"], "valid=" + str(valid), "SEEL=" + str(record["seel_nm"]), flush=True)
        if not valid:
            return 1
    state["status"] = "complete"
    write_json(state_path, state)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
