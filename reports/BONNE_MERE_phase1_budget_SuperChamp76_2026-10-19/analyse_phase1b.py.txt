import json, os, numpy as np
ABL = os.path.join(os.environ["TEMP"], "bm_harness", "abl"); SEEDS = [20261019501, 20261019502, 20261019503]
design = json.load(open(os.path.join(os.environ["TEMP"], "bm_replay/study/design/retenu_96_saphir/design.json"), encoding="utf-8"))
nom = np.array(design["thickness_nm"])
def get(cfg, i):
    z = [np.load(os.path.join(ABL, f"{cfg}_s{s}", "component_scores.npz")) for s in SEEDS]
    v = np.concatenate([q[f"valid_{i}"] for q in z])
    return v, np.concatenate([q[f"thickness_nm_{i}"] for q in z]), np.concatenate([q[f"seel_nm_{i}"] for q in z])
v, th, s = get("ref", 1); e = th[v] - nom; y = s[v] ** 2
X = np.hstack([e, e ** 2, np.ones((len(e), 1))]); c, *_ = np.linalg.lstsq(X, y, rcond=None)
terms = e * c[:96] + e ** 2 * c[96:192]; pred = terms.sum(1)
share = np.array([np.cov(terms[:, j], pred)[0, 1] for j in range(96)]) / np.var(y)
o = np.argsort(share)[::-1]
print("covariance shares (sum = R2 =", round(share.sum(), 3), "):", ", ".join(f"op {j+3} {100*share[j]:.1f}%" for j in o[:10]))
print("most negative (compensating):", ", ".join(f"op {j+3} {100*share[j]:.1f}%" for j in o[-4:]))
print("corr(e58, e59) =", round(np.corrcoef(e[:, 55], e[:, 56])[0, 1], 3), "| corr(e64, e65) =", round(np.corrcoef(e[:, 61], e[:, 62])[0, 1], 3))
print("sd(e) of the top layers by config at 0.05:")
for cfg in ("ref", "corr0", "slit0", "anch0", "n100", "all0"):
    vv, t, _ = get(cfg, 1); ee = t[vv] - nom
    print(f"  {cfg:<6}", "  ".join(f"op{j+3} {ee[:, j].std():.3f}" for j in (62, 55, 61, 74)), f" | mean bias op58 {ee[:, 55].mean():+.3f}")
print("stops by noise level, 3000 draws each:")
for cfg in ("ref", "corr0", "slit0", "anch0", "n100", "all0"):
    print(f"  {cfg:<6}", [int((~get(cfg, i)[0]).sum()) for i in range(3)])
