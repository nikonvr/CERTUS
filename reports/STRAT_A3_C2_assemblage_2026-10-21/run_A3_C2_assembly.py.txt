"""Re-score existing draws with the exact index stream of their cached noise level."""

import argparse
import hashlib
import io
import json
import os
import re
import shutil
import subprocess
import sys
from contextlib import redirect_stdout
from datetime import datetime, timezone
from pathlib import Path

ROOT = next(parent for parent in Path(__file__).resolve().parents if (parent / "scripts/bench_examples.py").is_file())
FOLDER = ROOT / "reports/STRAT_A3_C2_assemblage_2026-10-21"
SOURCE = ROOT / "reports/STRAT_A3_intervalles_reprise_2026-10-21"


def read(path):
    return json.loads(path.read_text(encoding="utf-8-sig"))


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def write(path, value):
    temporary = path.with_suffix(path.suffix + ".tmp")
    temporary.write_text(json.dumps(value, indent=2) + "\n", encoding="utf-8")
    os.replace(temporary, path)


parser = argparse.ArgumentParser()
parser.add_argument("--prepare", action="store_true")
parser.add_argument("--child", type=int, choices=(42, 101, 202))
parser.add_argument("--slot")
parser.add_argument("--resume-interrupted", action="store_true")
args = parser.parse_args()
if not args.prepare:
    parser.error("A3_C2_BLOCKED: differing interval normalization domains; this prototype must not score or launch a child")
if args.prepare:
    FOLDER.mkdir(exist_ok=False)
    (FOLDER / "run_A3_C2_assembly.py.txt").write_bytes(Path(__file__).read_bytes())
    prior = read(SOURCE / "intervalles_etat.json")
    assert prior["status"] == "complete"
    noise_line = next(line for line in (ROOT / "scratch/A3_noise_readout_20261021.log").read_text(encoding="utf-8-sig").splitlines() if line.startswith("A3_NOISE_READOUT="))
    noise = json.loads(noise_line.split("=", 1)[1])
    selected = min(range(len(noise["levels"])), key=lambda index: abs(noise["levels"][index] - 1.0))
    assert selected == 2 and noise["levels"] == [0.025, 0.05, 0.1]
    inputs = {}
    for seed in (42, 101, 202):
        item = next(item for item in prior["attempts"] if item["key"] == f"s{seed:03d}_assembly" and item["valid"])
        target = FOLDER / "inputs" / f"s{seed:03d}"
        shutil.copytree(ROOT / item["slot"] / "cache", target)
        for path in target.iterdir():
            assert sha(path) == sha(ROOT / item["slot"] / "cache" / path.name)
            inputs[path.relative_to(FOLDER).as_posix()] = sha(path)
    write(FOLDER / "protocole.json", {"prepared_utc": datetime.now(timezone.utc).isoformat(), "input_head": prior["head"],
          "seeds": [42, 101, 202], "cache_noise_index": selected, "noise_readout": noise,
          "inputs_sha256": inputs, "scientific_sources_sha256": read(SOURCE / "sources_sha256.json"),
          "old_assembly_scientifically_valid": False, "new_growth_draws": False, "independent_confirmation": False,
          "method": "Unchanged classer_partitions.main; child-scoped index-seed alias routes its request for index 1 to the actual cached index 2",
          "deadline_utc": "2026-10-21T08:00:00+00:00"})
    print("A3_C2=PREPARED, original thickness arrays unchanged; three re-scorings only")
    sys.exit(0)
protocol = read(FOLDER / "protocole.json")
if args.child:
    os.chdir(ROOT)
    sys.path[:0] = [str(ROOT), str(ROOT / "scripts")]
    import bench_examples as bench

    bench.qapp()
    __import__("CERTUS_STRAT")
    import classer_partitions as assembly
    from certus.core import certus_strat_robustness as robustness

    original_seed = robustness._index_stream_seed
    calls = []

    def routed(seed, noise_index):
        assert seed == args.child and noise_index == 1
        value = original_seed(seed, protocol["cache_noise_index"])
        calls.append({"base_seed": seed, "requested_noise_index": noise_index, "growth_and_score_noise_index": 2, "index_stream_seed": value})
        return value

    robustness._index_stream_seed = routed
    slot = ROOT / args.slot
    sys.argv = ["classer_partitions.py", "--cache-dir", str(slot / "cache"), "--mode", "fast", "--graine", str(args.child)]
    capture = io.StringIO()
    with redirect_stdout(capture):
        code = assembly.main()
    robustness._index_stream_seed = original_seed
    printed = capture.getvalue()
    print(printed, end="", flush=True)
    assert code == 0 and len(calls) == 1
    matches = re.findall(r"^\s*\d+\s+4\s+0-22\s+22-42\s+42-76\s+76-99\s+([0-9.]+)\s", printed, re.MULTILINE)
    assert len(matches) <= 1
    write(slot / "result.json", {"measured": True, "valid": len(matches) == 1,
          "seel_nm_printed": float(matches[0]) if matches else None, "printed_resolution_nm": 0.001,
          "index_stream_calls": calls, "new_draws": False})
    sys.stdout.flush()
    sys.stderr.flush()
    os._exit(0)
frozen = FOLDER / "run_A3_C2_assembly.py.txt"
assert frozen.read_bytes() == Path(__file__).read_bytes()
for name, expected in protocol["scientific_sources_sha256"].items():
    assert sha(ROOT / name) == expected, name
for name, expected in protocol["inputs_sha256"].items():
    assert sha(FOLDER / name) == expected, name
head = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True).strip()
state_path = FOLDER / "etat.json"
state = read(state_path) if state_path.exists() else {"head": head, "attempts": [], "status": "prepared"}
assert state["head"] == head and (not state["attempts"] or args.resume_interrupted)
for seed in protocol["seeds"]:
    if any(item["seed"] == seed and item.get("measured") for item in state["attempts"]):
        continue
    if datetime.now(timezone.utc) >= datetime.fromisoformat(protocol["deadline_utc"]):
        state["status"] = "deadline_reached"
        write(state_path, state)
        sys.exit(0)
    assert subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True).strip() == head
    assert not subprocess.check_output(["git", "status", "--porcelain", "--untracked-files=no"], cwd=ROOT, text=True).strip()
    attempt = 1 + sum(item["seed"] == seed for item in state["attempts"])
    slot = FOLDER / f"s{seed:03d}_a{attempt:02d}"
    shutil.copytree(FOLDER / "inputs" / f"s{seed:03d}", slot / "cache")
    record = {"seed": seed, "attempt": attempt, "slot": slot.relative_to(ROOT).as_posix(), "measured": False}
    state["attempts"].append(record)
    state.update(status="running", controller_pid=os.getpid())
    write(state_path, state)
    print("START", seed, flush=True)
    with (slot / "raw.log").open("xb") as output:
        process = subprocess.Popen([sys.executable, str(frozen), "--child", str(seed), "--slot", record["slot"]], cwd=ROOT,
                                   env=dict(os.environ, QT_QPA_PLATFORM="offscreen", PYTHONIOENCODING="utf-8"), stdout=output, stderr=subprocess.STDOUT)
        record["child_pid"] = process.pid
        write(state_path, state)
        exit_code = process.wait()
    result = read(slot / "result.json") if (slot / "result.json").exists() else {}
    record.update(exit_code=exit_code, result=result, measured=exit_code == 0 and result.get("measured", False), raw_sha256=sha(slot / "raw.log"))
    state["status"] = "case_complete" if record["measured"] else "failed"
    write(state_path, state)
    print("END", seed, "measured=" + str(record["measured"]), flush=True)
    if not record["measured"]:
        sys.exit(1)
state["status"] = "complete"
write(state_path, state)
