"""Index-error share of the STRAT winner and of the joint-search plan: same plans, same draws, index corridor on / off.

python -X utf8 index_share.py COMP       seeds 20261022301-303, 1000 draws each; share = 1 - RMSE2(off) / RMSE2(on).
"""
import json
import os
import subprocess
import sys

import numpy as np

COMP = sys.argv[1]
HERE = os.path.join(os.environ["TEMP"], "r75")
V = os.path.join(HERE, "val")
REPO = r"D:\drivfl\01_Recherche_Enseignement\02_Couches_Minces_Optique\couches minces 2026\CERTUS\certus0310"
SEEDS = (20261022301, 20261022302, 20261022303)
plans = {"winner": json.load(open(os.path.join(V, f"winner_{COMP}.json"))),
         "joint": json.load(open(os.path.join(V, f"modelam_{COMP}.json")))["plan"]}
jobs = [dict(name=f"{n}__{ic}__{s}", plan=p, seed=s, runs=1000, save=True, override={"index_corridor": ic})
        for n, p in plans.items() for ic in (0.005, 0.0) for s in SEEDS]
jf, out = os.path.join(V, f"idx_{COMP}.json"), os.path.join(V, f"idx_{COMP}.jsonl")
json.dump(jobs, open(jf, "w"))
if os.path.exists(out):
    os.remove(out)
subprocess.run([sys.executable, "-X", "utf8", os.path.join(HERE, "r75_scan.py"), "--repo", REPO, "--jobs", jf, "--out", out,
                "--comp", COMP, "--mode", "deep"], check=True, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
z = {}
for line in open(out, encoding="utf-8"):
    r = json.loads(line)
    a = np.load(os.path.join(V, f"idx_{COMP}_arrays", r["name"] + ".npz"))
    z[r["name"]] = a
res = {}
for n in plans:
    m = {}
    for ic in (0.005, 0.0):
        r2 = np.concatenate([z[f"{n}__{ic}__{s}"]["rmse_1"][~z[f"{n}__{ic}__{s}"]["crashed_1"]] ** 2 for s in SEEDS])
        m[ic] = float(r2.mean())
    res[n] = dict(rmse2_on=m[0.005], rmse2_off=m[0.0], index_part=m[0.005] - m[0.0], share=1 - m[0.0] / m[0.005])
    print(f"{COMP} {n}: mean RMSE2 with index error {m[0.005]:.4e}, without {m[0.0]:.4e}, index part {m[0.005] - m[0.0]:.4e} "
          f"= {100 * res[n]['share']:.0f} % of the total", flush=True)
w, j = res["winner"], res["joint"]
print(f"{COMP}: index part joint / winner = {j['index_part'] / w['index_part']:.2f}; rest (no index) joint / winner = "
      f"{j['rmse2_off'] / w['rmse2_off']:.2f}", flush=True)
json.dump(res, open(os.path.join(V, f"index_share_{COMP}.json"), "w"), indent=1)
