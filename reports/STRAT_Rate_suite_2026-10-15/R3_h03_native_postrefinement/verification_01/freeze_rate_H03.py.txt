"""Freeze the new H03 protocol before verification and any measurement."""

import copy
from datetime import UTC, datetime

from rate_H03_guard import EXPERIMENT, LIMITS, POST, PROGRAM, ROOT, check_deadline, read, sha
from rate_R4_planner_resume_v3 import save

FILES = ["rate_native_H03_probe.py", "rate_H03_post_scorer.py", "rate_H03_guard.py", "rate_H03_post_planner.py",
         "rate_H03_parent_selection.py", "run_rate_H03.ps1", "rate_R4_candidates.py", "rate_R4_planner.py",
         "rate_R4_planner_resume_v3.py", "rate_native_pipeline_trace.py", "rate_joint_native_candidates_R3_wavelengths.py",
         "campagne_a2_runner.py"]


def main():
    check_deadline()
    if (EXPERIMENT / "protocol.json").exists():
        raise RuntimeError("H03 already frozen; inspect, never overwrite")
    original = read(PROGRAM / "R4_refinement/protocol.json")
    grid = PROGRAM / "R4_refinement/legal_grid.json"
    POST.mkdir(exist_ok=True)
    (POST / "legal_grid.json").write_bytes(grid.read_bytes())
    inputs = [PROGRAM / "reference_context.json", PROGRAM / "R4_refinement/complete_audit_01.json",
              PROGRAM / "R4_refinement/confirmation_01/complete_audit_01.json", grid,
              PROGRAM / "R3_h02_wavelengths/comparison_homogeneous.json",
              PROGRAM / "R0_control/control.json", PROGRAM / "autonomie_jusqu_20261016_0800/authorization.json"]
    protocol = {"status": "FROZEN_BEFORE_MEASUREMENT", "prepared_utc": datetime.now(UTC).isoformat(),
                "limits": LIMITS, "deadline_utc": check_deadline(), "source_sha256": original["source_sha256"],
                "control": original["control"], "depths": original["depths"], "anchors": copy.deepcopy(original["anchors"]),
                "R3_pair_index": 3, "max_R3_pairs": 3, "R4_prototype_closed": True,
                "native_both_arms": {"joint_widths": list(range(1, 8)), "fine_wavelengths": True, "joint_budget": 480,
                                     "Rate_budget": 240, "rounds": 6, "full_cap": 36, "shifts": [-2, -1, 0, 1, 2], "slit_nm": 2},
                "single_factor": "automatic same-arm post-refinement off/on",
                "selection": "explicit complete native consensus, best then different mask, signature ties; homogeneous same-arm parents before moves",
                "post_cost": "Additional cost of final native validation, max600 new proposals; no R4 budget reassignment or equal-budget speed claim",
                "transport": "Validated R0/h01/h02/R4 physical callbacks and Workerfloat32 unchanged",
                "script_sha256": {name: sha(ROOT / "scratch" / name) for name in FILES},
                "input_sha256": {str(p.relative_to(ROOT)).replace('\\', '/'): sha(p) for p in inputs}}
    save(EXPERIMENT / "protocol.json", protocol, immutable=True)
    (POST / "protocol.json").write_bytes((EXPERIMENT / "protocol.json").read_bytes())
    for name in FILES:
        target = EXPERIMENT / "sources" / (name + ".txt")
        target.parent.mkdir(exist_ok=True)
        with target.open("xb") as stream:
            stream.write((ROOT / "scratch" / name).read_bytes())
    print("H03_PROTOCOL_FROZEN_NO_MEASUREMENT")


if __name__ == "__main__":
    main()
