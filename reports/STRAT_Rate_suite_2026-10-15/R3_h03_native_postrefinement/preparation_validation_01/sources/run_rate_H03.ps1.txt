function Save-RateState {
    $rateState['updated_utc'] = (Get-Date).ToUniversalTime().ToString('o')
    $rateTemp = $rateStatePath + '.tmp'
    for ($rateRetry=0; $rateRetry -lt 20; $rateRetry++) {

            if (Test-Path -LiteralPath $rateStatePath) {
                [System.IO.File]::Replace($rateTemp,$rateStatePath,($rateStatePath + '.previous'))
            } else { [System.IO.File]::Move($rateTemp,$rateStatePath) }
            return
        } catch [System.IO.IOException] {
            if ($rateRetry -ge 19) { throw }
            $rateState['persistence_retries']++
            Start-Sleep -Milliseconds 100
        }
    }
}
