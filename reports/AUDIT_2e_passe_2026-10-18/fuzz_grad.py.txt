"""Sonde : gradients analytiques contre différences finies de l'ORACLE (oblique) et du coût de production (normal).

python -X utf8 fuzz_grad.py [N_oblique N_normal]   -- depuis la racine du dépôt
"""
import os, sys, time
import numpy as np

sys.stdout.reconfigure(encoding='utf-8')
sys.path.insert(0, os.path.join(os.getcwd(), 'tests', 'oracle'))
sys.path.insert(0, os.getcwd())
from tmm_reference import rt_stack_oblique  # noqa: E402
from certus.physics.gradient_oblique import compute_oblique_rt_and_grads_analytic, compute_gradient_all_layers_analytic  # noqa: E402
from certus.physics.gradient_utils import cost_numba_fast  # noqa: E402

NO, NN = (int(a) for a in (sys.argv[1:3] if len(sys.argv) >= 3 else (300, 200)))
rng = np.random.default_rng(777)
H = 0.01  # nm


def stack(rng, n_wls, n_layers):
    d = rng.uniform(20.0, 400.0, n_layers)
    n_re = rng.uniform(1.2, 3.5, n_layers)
    k = np.where(rng.random(n_layers) < 0.5, 0.0, 10 ** rng.uniform(-4, -0.3, n_layers))
    wls = np.sort(rng.uniform(300.0, 1600.0, n_wls))
    n_all = np.empty((n_wls, n_layers), dtype=np.complex128)
    for i, w in enumerate(wls):
        n_all[i] = np.maximum(1.05, n_re - 0.05 * (w - 550.0) / 1000.0) - 1j * k
    sub_re = rng.uniform(1.3, 3.0, n_wls)
    sub_k = np.zeros(n_wls) if rng.random() < 0.5 else 10 ** rng.uniform(-5, -1, n_wls)
    return wls, d, n_all, sub_re - 1j * sub_k


# --- oblique : dR/dd et dT/dd analytiques contre différences finies centrées de l'oracle ---
t0 = time.time()
worst = (0.0, None)
errs = []
shape_info = None
for c in range(NO):
    wls, d, n_all, n_sub = stack(rng, 3, int(rng.integers(1, 9)))
    ang = float(rng.choice([0.0, 15.0, 40.0, 65.0, 80.0]))
    is_s = bool(rng.random() < 0.5)
    var = np.arange(len(d), dtype=np.int64)
    R, T, dR, dT = compute_oblique_rt_and_grads_analytic(d, n_all, n_sub, wls, var, ang, is_s)
    shape_info = (R.shape, T.shape, dR.shape, dT.shape)
    for i, w in enumerate(wls):
        for j in range(len(d)):
            up, dn = d.copy(), d.copy()
            up[j] += H
            dn[j] -= H
            Ru, Tu = rt_stack_oblique(float(w), n_all[i], up, ang, is_s, n_inc=1.0 + 0j, n_sub=n_sub[i])
            Rd, Td = rt_stack_oblique(float(w), n_all[i], dn, ang, is_s, n_inc=1.0 + 0j, n_sub=n_sub[i])
            fdR, fdT = (Ru - Rd) / (2 * H), (Tu - Td) / (2 * H)
            gR = dR[i, j] if dR.shape == (len(wls), len(d)) else dR[j, i]
            gT = dT[i, j] if dT.shape == (len(wls), len(d)) else dT[j, i]
            e = max(abs(gR - fdR), abs(gT - fdT)) / (1.0 / w)  # écart rapporté à l'échelle naturelle 1/λ (par nm)
            errs.append(e)
            if e > worst[0]:
                worst = (e, (c, i, j, ang, 's' if is_s else 'p'))
e = np.array(errs)
print('oblique : formes (R,T,dR,dT) =', shape_info)
print('oblique : %d dérivées | max écart/(1/λ) = %.3e | > 1e-6 : %d | > 1e-4 : %d | > 1e-2 : %d | pire : %s | %.0f s' % (
    len(e), e.max(), (e > 1e-6).sum(), (e > 1e-4).sum(), (e > 1e-2).sum(), worst[1], time.time() - t0))

# --- normal : gradient du coût de production contre différences finies du coût ---
t0 = time.time()
rel = []
worstn = (0.0, None)
for c in range(NN):
    wls, d, n_all, n_sub = stack(rng, 6, int(rng.integers(1, 9)))
    has_back = bool(rng.random() < 0.5)
    if has_back:
        _, d_b, n_b, _ = stack(rng, 6, int(rng.integers(1, 5)))
        n_b = np.repeat(n_b[:1], len(wls), axis=0)
    else:
        d_b, n_b = np.zeros(0), np.zeros((len(wls), 0), dtype=np.complex128)
    tg = rng.uniform(0.05, 0.95, len(wls))
    wt = rng.uniform(0.2, 2.0, len(wls))
    var = np.arange(len(d), dtype=np.int64)
    val, grad = compute_gradient_all_layers_analytic(d, n_all, n_sub, wls, tg, wt, 1.0, has_back, n_b, d_b, var)
    cost = lambda x: cost_numba_fast(x, n_all, n_sub, wls, tg, wt, 1.0, has_back, n_b, d_b)  # noqa: E731
    c0 = cost(d)
    if abs(val - c0) > 1e-9 * max(1.0, abs(c0)):
        print('  valeur du coût différente (cas %d) : %.12g contre %.12g' % (c, val, c0))
    for j in range(len(d)):
        up, dn = d.copy(), d.copy()
        up[j] += 1e-3
        dn[j] -= 1e-3
        fd = (cost(up) - cost(dn)) / 2e-3
        scale = max(abs(fd), 1e-9)
        r = abs(grad[j] - fd) / scale
        rel.append(r)
        if r > worstn[0]:
            worstn = (r, (c, j, has_back))
r = np.array(rel)
print('normal : %d dérivées | max écart relatif = %.3e | > 1e-5 : %d | > 1e-3 : %d | > 1e-2 : %d | pire : %s | %.0f s' % (
    len(r), r.max(), (r > 1e-5).sum(), (r > 1e-3).sum(), (r > 1e-2).sum(), worstn[1], time.time() - t0))
