"""Verify the copied h02 replay after native work ends; never simulate a deposition."""

from __future__ import annotations

import ast
import copy
import hashlib
import json
import runpy
import sys
from datetime import UTC, datetime, timedelta
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
PROGRAM = ROOT / "reports/STRAT_Rate_suite_2026-10-15"
EXPERIMENT = PROGRAM / "R3_h02_wavelengths"


def main():
    authorization = PROGRAM / "autonomie_jusqu_20261016_0800/authorization.json"
    window = json.loads(authorization.read_text(encoding="utf-8"))
    deadline = datetime.fromisoformat(window["no_new_measurements_or_tests_after_utc"].replace("Z", "+00:00"))
    if datetime.now(UTC) >= deadline:
        raise RuntimeError("No new verification after the authorized morning cutoff")
    if (PROGRAM / "measurement.running").exists() or list((ROOT / "reports").rglob(".running")):
        raise RuntimeError("Do not start verification while a scientific measurement lives")
    output = EXPERIMENT / "replay_verification_01"
    output.mkdir(exist_ok=False)
    checks = []

    def reject(function, *args, **kwargs):
        try:
            function(*args, **kwargs)
        except RuntimeError as error:
            checks.append("Rejected: " + str(error))
        else:
            raise AssertionError("The invalid boundary case was accepted")

    guard = runpy.run_path(str(ROOT / "scratch/rate_R3_h02_replay_guard.py"))
    guard["check_deadline"](deadline.isoformat(), now=deadline - timedelta(microseconds=1))
    checks.append("Authorized instant immediately before cutoff accepted")
    reject(guard["check_deadline"], deadline.isoformat(), now=deadline)
    reject(guard["check_deadline"], deadline.isoformat(), now=deadline + timedelta(microseconds=1))
    reject(guard["check_deadline"], (deadline + timedelta(seconds=1)).isoformat())
    reject(guard["check_deadline"], deadline.replace(tzinfo=None).isoformat())
    for name in ("check_sources", "record_task", "keep_fixed_plans", "evaluate_fixed"):
        nodes = []
        for file in ("rate_fixed_replay_R3_h01.py", "rate_fixed_replay_R3_h02.py"):
            tree = ast.parse((ROOT / "scratch" / file).read_text(encoding="utf-8"))
            nodes.append(next(n for n in ast.walk(tree) if isinstance(n, ast.FunctionDef) and n.name == name))
        assert ast.dump(nodes[0], include_attributes=False) == ast.dump(nodes[1], include_attributes=False), name
        checks.append(name + " AST unchanged from the validated h01/R0 path")
    raw = json.loads((EXPERIMENT / "frozen_finalists.json").read_text(encoding="utf-8"))
    protocol = json.loads((EXPERIMENT / "replay_protocol.json").read_text(encoding="utf-8"))
    comparators = json.loads((PROGRAM / "R3_h01_widths/frozen_finalists.json").read_text(encoding="utf-8"))
    guard["validate_frozen_plans"](raw, protocol, comparators)
    anchors = [copy.deepcopy(plan) for strategy_id in protocol["comparator_ids"]
               for plan in raw["plans"] if plan["source_strategy_id"] == strategy_id]

    def fixture(plans):
        return {"plans": plans}, {**protocol, "unique_plans": len(plans), "actual_calls_required": 4 * len(plans)}

    for count in (4, 8):
        plans = copy.deepcopy(anchors)
        for index in range(count - 4):
            plan = copy.deepcopy(anchors[0])
            plan.update(source_strategy_id=98_000_000 + index, monochromator_resolution_nm=10.0 + index)
            plans.append(plan)
        test_raw, test_protocol = fixture(plans)
        guard["validate_frozen_plans"](test_raw, test_protocol, comparators)
        checks.append(f"Synthetic {count}-plan bound accepted without physical scoring")
        if count == 8:
            extra = copy.deepcopy(plans[-1])
            extra.update(source_strategy_id=99_000_000, monochromator_resolution_nm=99.0)
            reject(guard["validate_frozen_plans"], *fixture([*plans, extra]), comparators)
    reject(guard["validate_frozen_plans"], *fixture(anchors[:3]), comparators)
    duplicated_id = [*copy.deepcopy(anchors), copy.deepcopy(anchors[0])]
    reject(guard["validate_frozen_plans"], *fixture(duplicated_id), comparators)
    duplicate_plan = copy.deepcopy(anchors[0])
    duplicate_plan["source_strategy_id"] = 98_000_001
    reject(guard["validate_frozen_plans"], *fixture([*anchors, duplicate_plan]), comparators)
    changed = copy.deepcopy(anchors)
    changed[0]["expected_score_fixed_point"] *= 1.01
    reject(guard["validate_frozen_plans"], *fixture(changed), comparators)
    changed = copy.deepcopy(anchors)
    changed[0]["monochromator_resolution_nm"] = 2.1
    reject(guard["validate_frozen_plans"], *fixture(changed), comparators)
    reject(guard["validate_frozen_plans"], raw, {**protocol, "actual_calls_required": 0}, comparators)
    reference = json.loads((PROGRAM / "reference_context.json").read_text(encoding="utf-8"))
    sources = reference["source_sha256"]
    reject(guard["check_ready"], "R0", EXPERIMENT / "homogeneous_replay", sources)
    reject(guard["check_ready"], "R3_h02_replay", PROGRAM / "R0_control", sources)
    ready = guard["check_ready"]("R3_h02_replay", EXPERIMENT / "homogeneous_replay", sources,
                                  requested_deadline=deadline.isoformat())
    checks.append("Actual idle/source/authorization/protocol/quota guard GO; no reservation taken")
    worktree = Path(reference.get("worktree", reference.get("workspace_directory", "")))
    sys.path[:0] = [str(worktree), str(worktree / "scripts")]
    from certus.utils.certus_strat_context import _validate_strategy_blocks_contract

    for plan in raw["plans"]:
        valid, reason = _validate_strategy_blocks_contract(plan, 75)
        assert valid, (plan["nom"], reason)
    checks.append("All actual frozen complete plans pass the real75-layer mixed-plan contract")
    report = {"verdict": "GO", "verified_utc": datetime.now(UTC).isoformat(), "checks": checks,
              "physics_measured": False, "ready_proof": ready, "source_sha256": sources,
              "unique_plans": len(raw["plans"]), "required_physical_calls": len(raw["plans"]) * 4,
              "script_sha256": {file: hashlib.sha256((ROOT / "scratch" / file).read_bytes()).hexdigest()
                                for file in ("rate_fixed_replay_R3_h02.py", "rate_R3_h02_replay_guard.py",
                                             "verify_rate_R3_h02_replay.py")},
              "review": "Exact callback ASTs, explicit boundaries, full comparator metadata; manual review remains separate"}
    with (output / "verification.json").open("x", encoding="utf-8") as stream:
        json.dump(report, stream, ensure_ascii=False, indent=2)
    print(json.dumps({"verdict": "GO", "checks": len(checks), "physics_measured": False}))


if __name__ == "__main__":
    main()
