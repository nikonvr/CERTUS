"""Partition rule test: every Rate layer thicker than a cap goes optical, at its best single-flip wavelength.

python -X utf8 cap.py COMP OFFSET
Reads val/winner_COMP.json, val/nominal_COMP.json and val/scan_flip_COMP.jsonl (search seed). The Rate layers that
have a stop-free optical wavelength are ranked by nominal thickness; plans flip the N thickest (N = 1, 2, 4, 8, 16, 32,
all). All plans are judged against the winner on selection seeds (20261021101+OFFSET.., 3 x 1000 draws); the best
admissible N (no extra stop, gain > 2 SE, STRAT score not above the winner's) is then judged once on holdout seeds
(20261021201+OFFSET..). Same criterion as refine.py: paired mean RMSE^2 at the nominal noise.
"""
import json
import os
import subprocess
import sys

import numpy as np

COMP, OFF = sys.argv[1], int(sys.argv[2])
RUNS = int(os.environ.get("CAP_RUNS", "1000"))  # smoke tests only
TAG = f"cap_{COMP}" + ("" if RUNS == 1000 else "_smoke")
HERE = os.path.join(os.environ["TEMP"], "r75")
V = os.path.join(HERE, "val")
REPO = r"D:\drivfl\01_Recherche_Enseignement\02_Couches_Minces_Optique\couches minces 2026\CERTUS\certus0310"
SEL = tuple(20261021101 + OFF + i for i in range(3))
HOLD = tuple(20261021201 + OFF + i for i in range(3))
LOG = open(os.path.join(V, f"{TAG}.log"), "a", encoding="utf-8")


def say(*a):
    msg = " ".join(str(x) for x in a)
    print(msg, flush=True)
    LOG.write(msg + "\n")
    LOG.flush()


def run(tag, plans, seeds):
    jobs = [dict(name=f"{n}__{s}", plan=p, seed=s, runs=RUNS, save=True) for n, p in plans.items() for s in seeds]
    jf, out = os.path.join(V, f"rf_{TAG}_{tag}.json"), os.path.join(V, f"rf_{TAG}_{tag}.jsonl")
    json.dump(jobs, open(jf, "w"))
    if os.path.exists(out):
        os.remove(out)
    subprocess.run([sys.executable, "-X", "utf8", os.path.join(HERE, "r75_scan.py"), "--repo", REPO, "--jobs", jf, "--out", out,
                    "--comp", COMP, "--mode", "deep"], check=True, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
    res = {}
    for line in open(out, encoding="utf-8"):
        r = json.loads(line)
        n, s = r["name"].rsplit("__", 1)
        z = np.load(os.path.join(V, f"rf_{TAG}_{tag}_arrays", r["name"] + ".npz"))
        res.setdefault(n, {})[int(s)] = (z, r["robustness_score"])
    return res


def compare(res, ref, cand, seeds):
    d, stops, stops_ref, sc_ref, sc_c = [], 0, 0, [], []
    for s in seeds:
        zr, scr = res[ref][s]
        zc, scc = res[cand][s]
        stops += sum(int(zc[f"crashed_{i}"].sum()) for i in range(3))
        stops_ref += sum(int(zr[f"crashed_{i}"].sum()) for i in range(3))
        both = ~zr["crashed_1"] & ~zc["crashed_1"]
        d.append(zc["rmse_1"][both] ** 2 - zr["rmse_1"][both] ** 2)
        sc_ref.append(scr)
        sc_c.append(scc)
    d = np.concatenate(d)
    base = np.mean(np.concatenate([res[ref][s][0]["rmse_1"][~res[ref][s][0]["crashed_1"]] ** 2 for s in seeds]))
    return dict(d=float(d.mean()), se=float(d.std(ddof=1) / np.sqrt(len(d))), rel=float(d.mean() / base), stops=stops,
                stops_ref=stops_ref, score_ref=float(np.mean(sc_ref)), score=float(np.mean(sc_c)))


def flip(plan, k, lam):
    out = []
    for b in plan["blocks"]:
        s, e, w = b["start"], b["end"], float(b["wavelength"])
        if s <= k < e:
            if s < k:
                out.append(dict(start=s, end=k, wavelength=w))
            out.append(dict(start=k, end=k + 1, wavelength=float(lam)))
            if k + 1 < e:
                out.append(dict(start=k + 1, end=e, wavelength=w))
        else:
            out.append(dict(start=s, end=e, wavelength=w))
    q = dict(plan)
    q["blocks"] = out
    q["rate_layers"] = [r for r in plan["rate_layers"] if r != k]
    return q


def p95_seel(res, name, seeds):
    p95 = max(float(np.percentile(np.concatenate([res[name][s][0][f"rmse_{i}"] for s in seeds]), 95)) for i in range(3))
    return 2 * p95 ** 0.5


winner = json.load(open(os.path.join(V, f"winner_{COMP}.json")))
thick = json.load(open(os.path.join(V, f"nominal_{COMP}.json")))
best = {}
for line in open(os.path.join(V, f"scan_flip_{COMP}.jsonl"), encoding="utf-8"):
    r = json.loads(line)
    if r["name"] == "base" or any(x["crashed"] for x in r["per_noise"]):
        continue
    k, v = r["meta"]["layer"], r["per_noise"][1]["rmse2_mean_done"]
    if k not in best or v < best[k][0]:
        best[k] = (v, r["meta"]["lam"])
order = sorted(best, key=lambda k: -thick[k])
no_lam = sorted(set(winner["rate_layers"]) - set(best))
say(f"== {COMP}: {len(winner['rate_layers'])} Rate layers, {len(order)} with a stop-free optical wavelength; kept in Rate for lack of one: {no_lam}")
plans, caps = {"ref": winner}, {}
for n in sorted({x for x in (1, 2, 4, 8, 16, 32, len(order)) if 0 < x <= len(order)}):
    p = winner
    for k in order[:n]:
        p = flip(p, k, best[k][1])
    plans[f"N{n}"] = p
    caps[f"N{n}"] = thick[order[n - 1]]
res = run("sel", plans, SEL)
adm = []
for name in plans:
    if name == "ref":
        continue
    c = compare(res, "ref", name, SEL)
    ok = c["stops"] <= c["stops_ref"] and c["d"] < -2 * c["se"] and c["score"] <= c["score_ref"]
    say(f"   {name} (cap {caps[name]:.0f} nm): dRMSE2 {100 * c['rel']:+.1f} %, stops {c['stops']} vs {c['stops_ref']}, "
        f"SEEL {p95_seel(res, 'ref', SEL):.4f} -> {p95_seel(res, name, SEL):.4f} nm -> {'admissible' if ok else 'no'}")
    if ok:
        adm.append((c["d"], name))
final = None
if adm:
    name = min(adm)[1]
    r3 = run("hold", {"ref": winner, name: plans[name]}, HOLD)
    c = compare(r3, "ref", name, HOLD)
    final = dict(c, plan=name, cap_nm=caps[name], seel_ref=p95_seel(r3, "ref", HOLD), seel=p95_seel(r3, name, HOLD))
    say(f"   hold {name}: dRMSE2 {100 * c['rel']:+.1f} %, stops {c['stops']} vs {c['stops_ref']}, SEEL {final['seel_ref']:.4f} -> {final['seel']:.4f} nm")
    json.dump(plans[name], open(os.path.join(V, f"capped_{COMP}.json" if RUNS == 1000 else "capped_smoke.json"), "w"), indent=1)
else:
    say("   no admissible cap")
json.dump(dict(comp=COMP, order=order, caps=caps, final=final), open(os.path.join(V, f"{TAG}.json"), "w"), indent=1)
say("done")
