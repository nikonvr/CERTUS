def source(path):
    file = ROOT / path
    raw = file.read_bytes()
    SOURCES[path] = hashlib.sha256(raw).hexdigest()
    return ast.parse(raw.decode("utf-8-sig"), filename=str(file))


def extract(path, name, ns, *, decorators=False):
    nodes = source(path).body
    selected = None
    for part in name.split("."):
        selected = next(n for n in nodes if isinstance(n, (ast.FunctionDef, ast.ClassDef))
                        and n.name == part)
        nodes = selected.body
    selected = copy.deepcopy(selected)
    if not decorators:
        selected.decorator_list = []
    tree = ast.Module(body=[ast.ImportFrom(module="__future__",
        names=[ast.alias(name="annotations")], level=0), selected], type_ignores=[])
    ast.fix_missing_locations(tree)
    ns.setdefault("__name__", "__main__")
    exec(compile(tree, str(ROOT / path), "exec"), ns)
    return ns[selected.name]

def fake_modules(values):
    modules = {}
    for full, attrs in values.items():
        parts = full.split(".")
        for i in range(1, len(parts) + 1):
            name = ".".join(parts[:i])
            if name not in modules:
                mod = types.ModuleType(name)
                mod.__path__ = []
                modules[name] = mod
        modules[full].__dict__.update(attrs)
    return patch.dict(sys.modules, modules)

def normalization():
    observations = []
    for nk, weights in [
        (2.1 + 0j, np.ones(12) / 12),
        (2.1 + 0j, np.arange(1, 13, dtype=float) / 78),
        (2.1 - 0.02j, np.ones(12) / 12),
    ]:
        wl = np.linspace(400., 700., 12)
        ns = np.full(12, 1.52 + 0j)
        layers = np.full((12, 1), nk)
        needle = np.full((12, 1), 1.45 + 0j)
        target = np.full(12, .9)
        rows = []
        for scale in (1., 7.):
            k, depth, raw = NUM["needle_scan_cached"](
                wl, layers, needle, ns, np.array([40.]), target, weights * scale,
                10., .01, np.array([1]))
            assert k == 0 and np.isfinite(raw)
            ds = [depth, .01, 40. - depth]
            ys = np.array([REF["rt_stack"](w, [nk, 1.45 + 0j, nk], ds)[1] for w in wl])
            reference, count = NUM["compute_mse_vectorized"](ys, target, weights * scale)
            expected = float((weights * scale).sum() / count)
            ratio = float(raw / reference)
            assert np.isclose(ratio, expected, rtol=1e-7, atol=1e-10), (ratio, expected)
            rows.append({"scale": scale, "depth": float(depth), "scan_cost": float(raw),
                         "reference_cost": float(reference), "ratio": ratio})
        assert np.isclose(rows[1]["scan_cost"] / rows[0]["scan_cost"], 7.)
        assert np.isclose(rows[0]["reference_cost"], rows[1]["reference_cost"])
        observations.append({"n": str(nk), "weights": weights.tolist(), "results": rows})
    return observations

def math_checks():
    n, wl, h = 2.1 + 0j, 550., 1e-5
    actual = (REF["characteristic_matrix"](n, h, wl) - np.eye(2)) / h
    corrected = 2j*np.pi*n/wl * np.array([[0, 1/n], [n, 0]])
    old = corrected / n
    err = float(np.linalg.norm(actual - corrected))
    assert err < 1e-7 and np.linalg.norm(actual - old) > 1e-3
    ds = [42.75, 20., 42.75]
    original = [REF["rt_stack"](w, [2.+0j], [100.])[1] for w in (450., 550., 700.)]
    compensated = [REF["rt_stack"](w, [2.+0j, 1.45+0j, 2.+0j], ds)[1] for w in (450., 550., 700.)]
    delta = float(np.max(np.abs(np.array(original) - compensated)))
    assert np.isclose(2*100., 2*(ds[0]+ds[2]) + 1.45*ds[1]) and delta > 1e-3
    # An optical quadratic can overpower the concave gap penalty.
    total = lambda x: 2*(x-2.5)**2 + x*(5-x)
    assert total(2.5) < total(2.4) and total(2.5) < total(2.6)
    ts, tp, target = .8, .2, .5
    avg_cost = ((ts + tp)/2 - target)**2
    separate_cost = ((ts-target)**2 + (tp-target)**2)/2
    assert np.isclose(separate_cost - avg_cost, (ts-tp)**2/4)
    return {"needle_matrix_derivative_error": err,
            "same_optical_thickness_max_T_difference": delta,
            "gap_counterexample_local_min_nm": 2.5,
            "avg_target_cost": avg_cost, "two_separate_target_cost": separate_cost}
