import os, sys
import numpy as np

sys.stdout.reconfigure(encoding='utf-8')
sys.path.insert(0, os.getcwd())
from certus.physics import certus_colorimetry as cc  # noqa: E402

print('== 1. réflecteur parfait (R = 1, 380-780 nm) ==')
wl = cc.CIE_LAMBDA
xyz = cc.xyz_from_spectrum(wl, np.ones_like(wl))
lab = cc.xyz_to_lab(xyz)
print('XYZ =', np.round(xyz, 4), '| attendu D65/2° :', cc.XYZ_N, '| Lab =', np.round(lab, 4), '(attendu 100, 0, 0)')
half = cc.xyz_to_lab(cc.xyz_from_spectrum(wl, np.full_like(wl, 0.5)))
print('R = 0,5 : L* =', round(float(half[0]), 3), '(attendu 76.069 pour Y = 50)')

print('\n== 2. tables CIE 1931 2° (valeurs tabulées connues) ==')
ref = {380: (0.001368, 0.000039, 0.006450), 450: (0.3362, 0.0380, 1.7721), 500: (0.0049, 0.3230, 0.2720), 555: (0.5121, 1.0000, 0.0057), 600: (1.0622, 0.6310, 0.0008), 700: (0.0114, 0.0041, 0.0000)}
worst = 0.0
for w, (x, y, z) in ref.items():
    i = int(np.where(wl == w)[0][0])
    got = (cc.CIE_X[i], cc.CIE_Y[i], cc.CIE_Z[i])
    e = max(abs(got[0] - x), abs(got[1] - y), abs(got[2] - z))
    worst = max(worst, e)
    print(f'  {w} nm : x̄ȳz̄ = ({got[0]:.4f}, {got[1]:.4f}, {got[2]:.4f})  tabulé ({x}, {y}, {z})  écart max {e:.1e}')
print('  pire écart des 6 longueurs d\'onde : %.1e' % worst)

print('\n== 3. ΔE2000, paires de test de Sharma, Wu et Dalal (2005) ==')
pairs = [((50.0000, 2.6772, -79.7751), (50.0000, 0.0000, -82.7485), 2.0425), ((50.0000, 3.1571, -77.2803), (50.0000, 0.0000, -82.7485), 2.8615),
         ((50.0000, 2.8361, -74.0200), (50.0000, 0.0000, -82.7485), 3.4412), ((50.0000, 0.0000, 0.0000), (50.0000, -1.0000, 2.0000), 2.3669),
         ((50.0000, 2.5000, 0.0000), (50.0000, 0.0000, -2.5000), 4.3065), ((50.0000, 2.5000, 0.0000), (73.0000, 25.0000, -18.0000), 27.1492),
         ((50.0000, 2.5000, 0.0000), (61.0000, -5.0000, 29.0000), 22.8977), ((50.0000, 2.5000, 0.0000), (56.0000, -27.0000, -3.0000), 31.9030),
         ((60.2574, -34.0099, 36.2677), (60.4626, -34.1751, 39.4387), 1.2644), ((63.0109, -31.0961, -5.8663), (62.8187, -29.7946, -4.0864), 1.2630)]
worst = 0.0
for a, b, e in pairs:
    v = cc.delta_e_2000(np.array(a, dtype=float), np.array(b, dtype=float))
    worst = max(worst, abs(v - e))
    print(f'  {a} / {b} : {v:.4f}  attendu {e:.4f}  écart {v - e:+.1e}')
print('  pire écart : %.1e' % worst)

print('\n== 4. spectre qui ne couvre pas 380-780 nm, ou en ordre décroissant ==')
wl_part = np.arange(450.0, 651.0, 5.0)
R_part = np.full_like(wl_part, 0.8)
print('R = 0,8 sur 450-650 nm seulement : Y =', round(float(cc.xyz_from_spectrum(wl_part, R_part)[1]), 3), '| R = 0,8 sur 380-780 nm : Y =', round(float(cc.xyz_from_spectrum(wl, np.full_like(wl, 0.8))[1]), 3), ' -> hors plage, np.interp prolonge par la valeur du bord, sans avertissement')
R_ramp = np.linspace(0.1, 0.9, len(wl))
y_up = cc.xyz_from_spectrum(wl, R_ramp)
y_dn = cc.xyz_from_spectrum(wl[::-1], R_ramp[::-1])
print('rampe croissante en λ croissant :', np.round(y_up, 3), '| mêmes données en ordre décroissant :', np.round(y_dn, 3))
