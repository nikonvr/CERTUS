"""Isolation des leviers du Rate elargi : une ligne par artefact, puis la decomposition A -> C1 -> C2 -> B natif -> B.

    python -I analyse_isolation_leviers.py A=<defauts.json> C1=<budget240.json> C2=<budget240_six_tours.json> B=<elargi_raffinement.json>

Les memes champs que `scripts/resumer_artefact_sonde.py` (deposable = plantage < 5 %, SEEL = 2 racine(score)), plus le score de consensus
`robustness_score_consensus` (graines de consensus 41/42/43). « B natif » = le meilleur plan de B qui ne vient pas de RATE_POST, c'est-a-dire
le resultat de la recherche avant le raffinement final. Rien n'est ecrit : la sortie est lue a la main pour le bilan.
"""

from __future__ import annotations

import json
import math
import sys
from pathlib import Path

SEUIL = 0.05


def seel(score: float) -> float:
    return 2.0 * math.sqrt(score)


def lire(chemin: str) -> dict:
    a = json.loads(Path(chemin).read_text(encoding="utf-8"))
    strat = a.get("strategies") or []
    dep = [s for s in strat if float(s.get("crash_rate", 1.0)) < SEUIL]
    natifs = [s for s in dep if not str(s.get("origine", "")).startswith("RATE_POST")]
    return {"a": a, "strat": strat, "dep": dep, "nat": natifs}


def meilleur(plans: list[dict], cle: str) -> dict | None:
    plans = [s for s in plans if s.get(cle) is not None]
    return min(plans, key=lambda s: float(s[cle])) if plans else None


def decrire(s: dict | None, cle: str) -> str:
    if s is None:
        return "aucun"
    return (
        f"{float(s[cle]):.6f} id={s.get('id')} origine={str(s.get('origine'))[:22]} blocs={s.get('n_blocs')} "
        f"couches_rate={len(s.get('rate_layers') or [])} plantage={float(s.get('crash_rate', float('nan'))):.2%}"
    )


def pct(x: float, ref: float) -> str:
    return f"{(x - ref) / ref * 100:+.2f} %"


def main(argv: list[str]) -> int:
    if hasattr(sys.stdout, "reconfigure"):
        sys.stdout.reconfigure(encoding="utf-8", errors="replace")
    donnees: dict[str, dict] = {}
    for arg in argv:
        etiquette, _, chemin = arg.partition("=")
        d = lire(chemin)
        donnees[etiquette] = d
        a = d["a"]
        plantage_min = min((float(s.get("crash_rate", 1.0)) for s in d["strat"]), default=float("nan"))
        print(f"[{etiquette}] {Path(chemin).name}")
        print(
            f"  graine={a.get('seed')} surcharges={(a.get('config') or {}).get('overrides')} strategies={len(d['strat'])} "
            f"deposables={len(d['dep'])} plantage_min={plantage_min:.4f}"
        )
        b = meilleur(d["dep"], "score")
        print("  meilleur SEEL " + (f"{seel(float(b['score'])):.4f} nm  {decrire(b, 'score')}" if b else "aucun deposable"))
        print("  meilleur consensus     " + decrire(meilleur(d["dep"], "robustness_score_consensus"), "robustness_score_consensus"))
        print("  meilleur consensus nat " + decrire(meilleur(d["nat"], "robustness_score_consensus"), "robustness_score_consensus"))
    if not {"A", "C1", "C2", "B"} <= set(donnees):
        return 0

    def consensus(d: dict, cle: str) -> float:
        s = meilleur(d[cle], "robustness_score_consensus")
        return float(s["robustness_score_consensus"]) if s else float("nan")

    def seel_min(d: dict, cle: str) -> float:
        s = meilleur(d[cle], "score")
        return seel(float(s["score"])) if s else float("nan")

    etapes = [
        ("A  defauts de production", consensus(donnees["A"], "dep"), seel_min(donnees["A"], "dep")),
        ("C1 + budget Rate 240", consensus(donnees["C1"], "dep"), seel_min(donnees["C1"], "dep")),
        ("C2 + six tours", consensus(donnees["C2"], "dep"), seel_min(donnees["C2"], "dep")),
        ("B natif + 480 conjoints", consensus(donnees["B"], "nat"), seel_min(donnees["B"], "nat")),
        ("B + raffinement final", consensus(donnees["B"], "dep"), seel_min(donnees["B"], "dep")),
    ]
    ecart_c = etapes[0][1] - etapes[3][1]
    ecart_s = etapes[0][2] - etapes[3][2]
    print()
    print(f"{'etape':28s} {'consensus':>10s} {'pas':>9s} {'cumul':>9s} {'part A>Bnat':>12s} | {'SEEL nm':>8s} {'pas':>9s} {'cumul':>9s} {'part':>8s}")
    for i, (nom, c, s) in enumerate(etapes):
        if i == 0:
            print(f"{nom:28s} {c:10.6f} {'':>9s} {'':>9s} {'':>12s} | {s:8.4f}")
            continue
        pc, ps = etapes[i - 1][1], etapes[i - 1][2]
        part_c = f"{(pc - c) / ecart_c * 100:6.1f} %" if i <= 3 and ecart_c else ""
        part_s = f"{(ps - s) / ecart_s * 100:6.1f} %" if i <= 3 and ecart_s else ""
        print(
            f"{nom:28s} {c:10.6f} {pct(c, pc):>9s} {pct(c, etapes[0][1]):>9s} {part_c:>12s} | "
            f"{s:8.4f} {pct(s, ps):>9s} {pct(s, etapes[0][2]):>9s} {part_s:>8s}"
        )
    return 0


if __name__ == "__main__":
    raise SystemExit(main(sys.argv[1:]))
