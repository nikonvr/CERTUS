"""Verify R4 refusal/recovery contracts without simulating a deposition or consuming a slot."""

from __future__ import annotations

import ast
import copy
import json
import subprocess
import sys
from datetime import UTC, datetime, timedelta
from pathlib import Path

import rate_R4_guard as guard
import rate_R4_planner as planner


def main() -> None:
    guard.check_deadline()
    if (guard.PROGRAM / "measurement.running").exists() or list((guard.ROOT / "reports").rglob(".running")):
        raise RuntimeError("No verification during scientific measurement")
    output = guard.EXPERIMENT / "verification_02"
    output.mkdir(exist_ok=False)
    checks = []

    def reject(function, *args, **kwargs):
        try:
            function(*args, **kwargs)
        except RuntimeError as error:
            checks.append("Rejected: " + str(error))
        else:
            raise AssertionError("Invalid case accepted")

    deadline = datetime(2026, 10, 16, 6, tzinfo=UTC)
    guard.check_deadline(deadline.isoformat(), now=deadline - timedelta(microseconds=1))
    checks.append("Instant immediately before morning cutoff accepted")
    for now in (deadline, deadline + timedelta(seconds=1)):
        reject(guard.check_deadline, deadline.isoformat(), now=now)
    reject(guard.check_deadline, (deadline + timedelta(seconds=1)).isoformat())
    reject(guard.check_deadline, deadline.replace(tzinfo=None).isoformat())
    for name in ("check_sources", "record_task", "keep_fixed_plans", "evaluate_fixed"):
        nodes = []
        for file in ("rate_fixed_replay_R3_h02.py", "rate_fixed_replay_R4.py"):
            tree = ast.parse((guard.ROOT / "scratch" / file).read_text(encoding="utf-8"))
            nodes.append(next(n for n in ast.walk(tree) if isinstance(n, ast.FunctionDef) and n.name == name))
        assert ast.dump(nodes[0], include_attributes=False) == ast.dump(nodes[1], include_attributes=False), name
        checks.append(name + " AST identical to validated h02/R0 callback")
    protocol, state = planner.protocol_and_state()
    trusted = guard.read(guard.PROGRAM / "R3_h02_wavelengths/homogeneous_replay/control.json")
    trusted_plans = {p["source_strategy_id"]: p for p in trusted["plans"]}
    for anchor in protocol["anchors"]:
        expected = trusted_plans[anchor["source_strategy_id"]]
        assert "expected_score_fixed_point" in anchor
        for key in ("blocks", "rate_layers", "monochromator_resolution_nm", "witness_reset_layers", "expected_score_fixed_point"):
            assert anchor.get(key) == expected.get(key), key
    checks.append("All three anchors retain independently measured fixed points after branch initialization")
    manifest_file = Path(state["pending_group"])
    manifest, raw = guard.read(manifest_file), guard.read(manifest_file.parent / "plans.json")
    guard.validate_group(manifest, raw, protocol, state)
    checks.append("Actual charged group, three strict anchors and nine proposals accepted")
    for key, value in (("new_proposals", 1201), ("group_count", 145)):
        changed = copy.deepcopy(state)
        changed[key] = value
        reject(guard.validate_group, manifest, raw, protocol, changed)
    changed = copy.deepcopy(state)
    changed["branch_pass_counts"]["local_pass1"] = 301
    reject(guard.validate_group, manifest, raw, protocol, changed)
    changed = copy.deepcopy(raw)
    changed["plans"][0]["expected_score_fixed_point"] *= 1.01
    reject(guard.validate_group, manifest, changed, protocol, state)
    for change in ({"branch": "native"}, {"root_parent_id": 901054165}, {"expected_score_fixed_point": 0.0}):
        changed = copy.deepcopy(raw)
        changed["plans"][3].update(change)
        changed_manifest = {**manifest, "plans_sha256": guard.hashlib.sha256(json.dumps(changed, ensure_ascii=False, indent=2).encode()).hexdigest()}
        # The queue equality also protects the exact immutable provenance.
        reject(guard.validate_group, changed_manifest, changed, protocol, state)
    changed = copy.deepcopy(state)
    changed["branches"]["native"]["beam"] *= 2
    reject(guard.validate_group, manifest, raw, protocol, changed)
    entries = []
    for plan, score in zip(raw["plans"][3:6], (3.0, 1.0, 2.0), strict=True):
        entries.append({"plan": plan, "score": score})
    # Same-mask lower scores must not evict the absolute best when diversification runs.
    selected = planner.beam(entries)
    assert selected[0]["score"] == 1.0
    assert planner.beam(list(reversed(entries))) == selected
    checks.append("Stable beam retains absolute best and deterministic mask/plan diversity")
    before = guard.sha(guard.EXPERIMENT / "checkpoint.json")
    reject(planner.next_group)
    assert guard.sha(guard.EXPERIMENT / "checkpoint.json") == before
    checks.append("Pending group without artifact is refused without rewriting the checkpoint")
    script = (guard.ROOT / "scratch/run_rate_R4.ps1").read_text(encoding="utf-8")
    functions = script[script.index("function Save-RateR4State"):script.index("\nSave-RateR4State\ntry {")]
    harness = output / "powershell_persistence.ps1"
    def quote(value):
        return "'" + str(value).replace("'", "''") + "'"
    harness.write_text("$ErrorActionPreference='Stop'\n" +
                       f"$rateR4Root={quote(guard.ROOT)}\n$rateR4Python={quote(sys.executable)}\n" +
                       f"$rateR4StatePath={quote(output / 'persistence.json')}\n" +
                       "$rateR4State=[ordered]@{status='test';runs=@();current_pid=$null}\n" + functions +
                       "\nSave-RateR4State\n" +
                       f"Invoke-RateR4Child @('-c','\"print(1)\"') {quote(output)}\n" +
                       "Save-RateR4State\n", encoding="utf-8")
    shell = Path(guard.os.environ["SYSTEMROOT"]) / "System32/WindowsPowerShell/v1.0/powershell.exe"
    with (output / "powershell.stdout.log").open("wb") as stdout, (output / "powershell.stderr.log").open("wb") as stderr:
        completed = subprocess.run([str(shell), "-NoProfile", "-Command", harness.read_text(encoding="utf-8")],
                                   stdout=stdout, stderr=stderr, check=False)
    if completed.returncode != 0:
        raise RuntimeError("Real Windows PowerShell5.1 persistence/handle verification failed")
    persisted = guard.read(output / "persistence.json")
    assert len(persisted["runs"]) == 1
    assert persisted["runs"][0]["exit_code"] == 0
    assert persisted["current_pid"] is None
    assert (output / "persistence.json.previous").exists()
    checks.append("Actual PowerShell5.1: explicit atomic backup, retained handle, recorded exit0, no simulation")
    for name in planner.FILES:
        assert guard.sha(guard.ROOT / "scratch" / name) == protocol["script_sha256"][name]
    report = {"verdict": "GO", "checks": checks, "verified_utc": datetime.now(UTC).isoformat(),
              "script_sha256": protocol["script_sha256"], "physics_measured": False,
              "claims_consumed": 0, "verification_directory": str(output),
              "manual_review": "Exact callback ASTs, bounded queue, branch separation, checkpoint refusal, strict anchors"}
    planner.save(output / "verification.json", report, immutable=True)
    # The frozen guard's proof location remains stable; the failed attempt logs stay intact.
    planner.save(guard.EXPERIMENT / "verification_01/verification.json", report, immutable=True)
    proof = guard.check_ready(manifest["phase"], manifest_file.parent / "measurement", protocol["source_sha256"],
                              requested_deadline=deadline.isoformat())
    reject(guard.check_ready, "R0", manifest_file.parent / "measurement", protocol["source_sha256"])
    reject(guard.check_ready, manifest["phase"], guard.PROGRAM / "R0_control", protocol["source_sha256"])
    reject(guard.check_ready, manifest["phase"], manifest_file.parent / "measurement", {"bad": "source"})
    planner.save(output / "actual_readiness.json", {"verdict": "GO", "ready": proof, "extra_checks": checks[-3:]}, immutable=True)
    print(json.dumps({"verdict": "GO", "checks": len(checks), "physics_measured": False}))


if __name__ == "__main__":
    main()
