"""Plan searched at +-0.005 (A) against plan searched at +-0.010 (B), both judged at +-0.005 and +-0.010, plus the winner.

python -X utf8 judge_ic.py COMP      fresh seeds 20261022601-603, 1000 draws each; same draws for every plan (C2).
Pre-registered rule (2026-10-20): B is preferred on a design when, at +-0.010, B beats A beyond two standard errors
(paired mean RMSE^2 at the nominal noise), at +-0.005 B loses at most 2 % of SEEL against A, and B adds no stop.
"""
import json
import os
import subprocess
import sys

import numpy as np

COMP = sys.argv[1]
HERE = os.path.join(os.environ["TEMP"], "r75")
V = os.path.join(HERE, "val")
REPO = r"D:\drivfl\01_Recherche_Enseignement\02_Couches_Minces_Optique\couches minces 2026\CERTUS\certus0310"
SEEDS = (20261022601, 20261022602, 20261022603)
CORR = (0.005, 0.010)
plans = {"winner": json.load(open(os.path.join(V, f"winner_{COMP}.json"))),
         "A": json.load(open(os.path.join(V, f"modelam_{COMP}.json")))["plan"],
         "B": json.load(open(os.path.join(V, f"modelam_{COMP}_ic010.json")))["plan"]}
jobs = [dict(name=f"{n}__{ic}__{s}", plan=p, seed=s, runs=1000, save=True, override={"index_corridor": ic})
        for n, p in plans.items() for ic in CORR for s in SEEDS]
jf, out = os.path.join(V, f"jic_{COMP}.json"), os.path.join(V, f"jic_{COMP}.jsonl")
json.dump(jobs, open(jf, "w"))
if os.path.exists(out):
    os.remove(out)
subprocess.run([sys.executable, "-X", "utf8", os.path.join(HERE, "r75_scan.py"), "--repo", REPO, "--jobs", jf, "--out", out,
                "--comp", COMP, "--mode", "deep"], check=True, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
z = {}
for line in open(out, encoding="utf-8"):
    r = json.loads(line)
    z[r["name"]] = np.load(os.path.join(V, f"jic_{COMP}_arrays", r["name"] + ".npz"))


def arrs(n, ic):
    return [z[f"{n}__{ic}__{s}"] for s in SEEDS]


def seel(n, ic):
    a = arrs(n, ic)
    return 2 * max(float(np.percentile(np.concatenate([x[f"rmse_{i}"] for x in a]), 95)) for i in range(3)) ** 0.5


def stops(n, ic):
    return sum(int(x[f"crashed_{i}"].sum()) for x in arrs(n, ic) for i in range(3))


def paired(ref, new, ic):
    d = []
    for r, c in zip(arrs(ref, ic), arrs(new, ic)):
        both = ~r["crashed_1"] & ~c["crashed_1"]
        d.append(c["rmse_1"][both] ** 2 - r["rmse_1"][both] ** 2)
    d = np.concatenate(d)
    base = np.mean(np.concatenate([r["rmse_1"][~r["crashed_1"]] ** 2 for r in arrs(ref, ic)]))
    return float(d.mean() / base), float(d.std(ddof=1) / np.sqrt(len(d)) / base)


res = {}
for ic in CORR:
    for n in plans:
        res[f"{n}@{ic}"] = dict(seel=seel(n, ic), stops=stops(n, ic))
    rel, se = paired("A", "B", ic)
    res[f"B_vs_A@{ic}"] = dict(rel=rel, se=se)
    print(f"{COMP} index +-{ic}: SEEL winner {res[f'winner@{ic}']['seel']:.4f}, A {res[f'A@{ic}']['seel']:.4f}, B {res[f'B@{ic}']['seel']:.4f} nm; "
          f"stops {stops('winner', ic)}/{stops('A', ic)}/{stops('B', ic)}; B vs A dRMSE2 {100 * rel:+.1f} % (SE {100 * se:.1f} %)", flush=True)
r10, r05 = res["B_vs_A@0.01"], res["B_vs_A@0.005"]
cost = res["B@0.005"]["seel"] / res["A@0.005"]["seel"] - 1
prefer_b = (r10["rel"] < -2 * r10["se"]) and cost <= 0.02 and stops("B", 0.005) <= stops("A", 0.005) and stops("B", 0.01) <= stops("A", 0.01)
res["cost_at_0.005"] = cost
res["prefer_B"] = bool(prefer_b)
print(f"{COMP}: cost of searching at +-0.010 when the index is at +-0.005: {100 * cost:+.1f} % SEEL; rule -> "
      f"{'search at +-0.010' if prefer_b else 'keep +-0.005'}", flush=True)
json.dump(res, open(os.path.join(V, f"judge_ic_{COMP}.json"), "w"), indent=1)
