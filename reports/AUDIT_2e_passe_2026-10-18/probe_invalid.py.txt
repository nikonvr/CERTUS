"""NaN / inf dans chaque entrée de chaque point d'entrée public : lève, rend NaN, ou rend du fini plausible (MAUVAIS) ?"""
import os, sys
import numpy as np

sys.stdout.reconfigure(encoding='utf-8')
sys.path.insert(0, os.getcwd())
from certus.physics.certus_tmm_matrix import calculate_RT_vectorized_real, calc_spectrum_full_exact  # noqa: E402
from certus.physics.certus_tmm_oblique import calc_spectrum_oblique_vectorized, calc_spectrum_oblique_backside_vectorized, calc_spectrum_full_oblique_exact  # noqa: E402
from certus.physics.gradient_utils import cost_numba_fast  # noqa: E402

wls = np.linspace(450.0, 650.0, 9)
d = np.array([80.0, 120.0])
n = np.tile(np.array([2.1 - 0.001j, 1.46 + 0j]), (9, 1))
ns = np.full(9, 1.52 + 0j)
db = np.array([100.0])
nb = np.tile(np.array([1.38 + 0j]), (9, 1))
tg = np.full(9, 0.5)
wt = np.ones(9)
nb0, db0 = np.zeros((9, 0), dtype=np.complex128), np.zeros(0)


def outcome(fn):
    try:
        out = fn()
    except Exception as e:  # noqa: BLE001
        return 'LÈVE', type(e).__name__
    arrs = out if isinstance(out, tuple) else (out,)
    flat = np.concatenate([np.atleast_1d(np.asarray(a, dtype=float)).ravel() for a in arrs])
    if np.all(np.isnan(flat)):
        return 'NaN', ''
    if np.any(np.isnan(flat)) or np.any(np.isinf(flat)):
        return 'PARTIEL', f'{np.isnan(flat).sum()} NaN / {flat.size}'
    return 'FINI', '(' + ', '.join(f'{x:.4g}' for x in flat[:4]) + ' ...)'


def corrupt(arr, value, pos=0):
    a = np.array(arr, copy=True)
    a.flat[pos] = value
    return a


entries = {
    'calculate_RT_vectorized_real': lambda W, D, N, S: calculate_RT_vectorized_real(D, N, S, W, with_backside=False),
    'calculate_RT_vectorized_real+arrière': lambda W, D, N, S: calculate_RT_vectorized_real(D, N, S, W, with_backside=True),
    'oblique_vectorized 0°': lambda W, D, N, S: calc_spectrum_oblique_vectorized(W, N, D, S, 0.0, 's'),
    'oblique_vectorized 30°': lambda W, D, N, S: calc_spectrum_oblique_vectorized(W, N, D, S, 30.0, 'p'),
    'oblique_backside_vectorized 30°': lambda W, D, N, S: calc_spectrum_oblique_backside_vectorized(W, N, D, S, 30.0, 's'),
    'full_oblique_exact 30°': lambda W, D, N, S: calc_spectrum_full_oblique_exact(W, D, N, db, nb, S, 30.0, True),
    'calc_spectrum_full_exact': lambda W, D, N, S: calc_spectrum_full_exact(W, D, N, db, nb, S),
    'cost_numba_fast': lambda W, D, N, S: cost_numba_fast(D, N, S, W, tg, wt, 0.01, False, nb0, db0),
}
inputs = ('wls', 'd', 'n_layers', 'n_sub')
bad = []
tally = {}
for name, fn in entries.items():
    base = outcome(lambda: fn(wls, d, n, ns))
    assert base[0] == 'FINI', (name, base)
    for which in inputs:
        for val in (np.nan, np.inf, -np.inf):
            args = {'wls': wls, 'd': d, 'n_layers': n, 'n_sub': ns}
            args[which] = corrupt(args[which], val)
            r = outcome(lambda: fn(args['wls'], args['d'], args['n_layers'], args['n_sub']))
            tally.setdefault(name, {}).setdefault(r[0], 0)
            tally[name][r[0]] += 1
            if r[0] in ('FINI', 'PARTIEL'):
                bad.append((name, which, val, r))
print('%-40s %s' % ('point d\'entrée', 'issues sur 12 entrées corrompues (4 entrées × NaN/+inf/-inf)'))
for name, t in tally.items():
    print('%-40s %s' % (name, dict(t)))
print('\nMAUVAIS (fini ou partiel malgré une entrée non finie) :', len(bad))
for name, which, val, r in bad[:30]:
    print(f'   {name:<38} {which:<9} {val!s:<5} -> {r[0]} {r[1]}')
