function Save-RateState {
    $rateState['updated_utc'] = (Get-Date).ToUniversalTime().ToString('o')
    $rateTemp = $rateStatePath + '.tmp'
    for ($rateRetry=0; $rateRetry -lt 20; $rateRetry++) {
        try {
            [System.IO.File]::WriteAllText($rateTemp,($rateState | ConvertTo-Json -Depth 12),[System.Text.UTF8Encoding]::new($false))
            if (Test-Path -LiteralPath $rateStatePath) {
                [System.IO.File]::Replace($rateTemp,$rateStatePath,($rateStatePath + '.previous'))
            } else { [System.IO.File]::Move($rateTemp,$rateStatePath) }
            return
        } catch [System.IO.IOException] {
            if ($rateRetry -ge 19) { throw }
            $rateState['persistence_retries']++
            Start-Sleep -Milliseconds 100
        }
    }
}
function Assert-RateDeadline {
    if ((Get-Date).ToUniversalTime() -ge [datetime]::Parse('2026-10-16T06:00:00Z').ToUniversalTime()) { throw 'Authorized morning cutoff reached; no new child' }
}
function Invoke-RateChild($rateArguments,$rateLogDirectory) {
    Assert-RateDeadline
    New-Item -ItemType Directory -Path $rateLogDirectory -ErrorAction Stop | Out-Null
    $rateProcess = Start-Process -FilePath $ratePython -ArgumentList $rateArguments -WorkingDirectory $rateRoot -WindowStyle Hidden -PassThru -RedirectStandardOutput (Join-Path $rateLogDirectory 'stdout.log') -RedirectStandardError (Join-Path $rateLogDirectory 'stderr.log')
    $rateHandle = $rateProcess.Handle
    $rateProcess.ProcessorAffinity = [IntPtr]16777215
    $rateProcess.PriorityClass = 'Normal'
    $rateState['current_pid'] = $rateProcess.Id
    Save-RateState
    $rateProcess.WaitForExit()
    $rateProcess.Refresh()
    $rateState['runs'] += [ordered]@{pid=$rateProcess.Id;exit_code=$rateProcess.ExitCode;log_directory=$rateLogDirectory;ended_utc=(Get-Date).ToUniversalTime().ToString('o')}
    $rateState['current_pid']=$null
    Save-RateState
    if ($null -eq $rateProcess.ExitCode -or $rateProcess.ExitCode -ne 0) { throw ('Child failed: ' + $rateProcess.Id + ', exit=' + $rateProcess.ExitCode) }
}
