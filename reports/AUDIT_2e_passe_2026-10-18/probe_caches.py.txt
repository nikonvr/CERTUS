"""C23 (caches qui rendent le même tableau modifiable) et C24 (cache de la matrice de base de la spline RE, clé arrondie à 1 nm)."""
import os
import sys

import numpy as np

sys.stdout.reconfigure(encoding='utf-8')
sys.path.insert(0, os.getcwd())

print('--- C23 : la valeur rendue est-elle partagée et modifiable ? ---')
from certus.core._certus_physics_impl import get_n_substrate_array_by_id  # noqa: E402
from certus.physics.certus_optical_models import get_nk_from_spline  # noqa: E402

w = np.linspace(400.0, 800.0, 5)
a = get_n_substrate_array_by_id(1, w)
b = get_n_substrate_array_by_id(1, w.copy())
print('substrat : même objet =', a is b, '| modifiable =', a.flags.writeable)
ref = float(a[0])
a[0] = 99.0
c = get_n_substrate_array_by_id(1, w.copy())
print('substrat : après a[0] = 99, un nouvel appel rend n[0] =', float(c[0]), '(attendu', round(ref, 5), ') ->', 'CACHE CORROMPU' if float(c[0]) == 99.0 else 'sain')
a[0] = ref
knots = np.array([400.0, 500.0, 600.0, 700.0, 800.0])
p = np.concatenate([np.array([1.5, 1.52, 1.54, 1.56, 1.58]), np.zeros(5)])
n1, k1 = get_nk_from_spline(p, knots, w, True)
n2, k2 = get_nk_from_spline(p.copy(), knots.copy(), w.copy(), True)
print('spline : même objet =', n1 is n2, '| modifiable =', n1.flags.writeable)
n0 = float(n1[0])
n1[0] = 99.0
n3, _ = get_nk_from_spline(p.copy(), knots.copy(), w.copy(), True)
print('spline : après n[0] = 99, un nouvel appel rend n[0] =', float(n3[0]), '(attendu', round(n0, 5), ') ->', 'CACHE CORROMPU' if float(n3[0]) == 99.0 else 'sain')
n1[0] = n0

print('\n--- C24 : matrice de base de la spline RE, grille décalée de 0,4 nm ---')
import certus.utils.certus_re_math as m  # noqa: E402

knots = np.linspace(400.0, 800.0, 6)
g1 = np.arange(450.0, 751.0, 1.0)
g2 = g1 + 0.4
m._re_bmat_cache.clear()
B1 = m.re_compute_spline_basis_matrix(knots, g1)
B2 = m.re_compute_spline_basis_matrix(knots, g2)
print('B2 est le même objet que B1 (cache) :', B2 is B1)
m._re_bmat_cache.clear()
B2_direct = m.re_compute_spline_basis_matrix(knots, g2)
dj = np.linspace(2.3, 2.0, len(knots))
print('écart max entre la matrice du cache et la matrice exacte : %.3e' % np.max(np.abs(B2 - B2_direct)))
print('effet sur n(lambda) : max |dS| = %.3e ; pente max de S = %.3e par nm' % (np.max(np.abs(B2 @ dj - B2_direct @ dj)), np.max(np.abs(np.gradient(B2_direct @ dj, g2)))))
