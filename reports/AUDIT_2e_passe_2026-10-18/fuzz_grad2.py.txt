"""Cohérence valeur/gradient du coût DESIGN en incidence normale : qui diffère, et sous quelle condition ?"""
import collections, os, sys
import numpy as np

sys.stdout.reconfigure(encoding='utf-8')
sys.path.insert(0, os.getcwd())
from certus.physics.gradient_oblique import compute_gradient_all_layers_analytic  # noqa: E402
from certus.physics.gradient_utils import cost_numba_fast  # noqa: E402

rng = np.random.default_rng(777)
rows = []


def mk(rng, n_wls, n_layers, absorbing_layers, absorbing_sub):
    d = rng.uniform(20.0, 400.0, n_layers)
    wls = np.sort(rng.uniform(300.0, 1600.0, n_wls))
    n_re = rng.uniform(1.3, 3.0, n_layers)
    k = 10 ** rng.uniform(-4, -0.5, n_layers) if absorbing_layers else np.zeros(n_layers)
    n_all = np.tile((n_re - 1j * k)[None, :], (n_wls, 1))
    sub = np.full(n_wls, 1.52 + 0j)
    if absorbing_sub:
        sub = sub - 1j * 10 ** rng.uniform(-5, -1)
    return wls, d, n_all, sub


for c in range(400):
    has_back = bool(c % 2)
    lay_abs = bool((c // 2) % 2)
    sub_abs = bool((c // 4) % 2)
    wnorm = bool((c // 8) % 2)
    wls, d, n_all, n_sub = mk(rng, 6, int(rng.integers(1, 7)), lay_abs, sub_abs)
    if has_back:
        _, d_b, n_b, _ = mk(rng, 6, int(rng.integers(1, 4)), lay_abs, False)
        n_b = np.tile(n_b[0][None, :], (len(wls), 1))
    else:
        d_b, n_b = np.zeros(0), np.zeros((len(wls), 0), dtype=np.complex128)
    tg = rng.uniform(0.05, 0.95, len(wls))
    wt = rng.uniform(0.5, 1.0, len(wls)) if wnorm else rng.uniform(0.2, 2.0, len(wls))
    if wnorm:
        wt = wt / wt.mean()
    var = np.arange(len(d), dtype=np.int64)
    val, grad = compute_gradient_all_layers_analytic(d, n_all, n_sub, wls, tg, wt, 1.0, has_back, n_b, d_b, var)
    cost = lambda x: cost_numba_fast(x, n_all, n_sub, wls, tg, wt, 1.0, has_back, n_b, d_b)  # noqa: E731
    c0 = cost(d)
    valbad = abs(val - c0) > 1e-9 * max(1.0, abs(c0))
    worst = 0.0
    for j in range(len(d)):
        up, dn = d.copy(), d.copy()
        up[j] += 1e-3
        dn[j] -= 1e-3
        fd = (cost(up) - cost(dn)) / 2e-3
        worst = max(worst, abs(grad[j] - fd) / max(abs(fd), 1e-9))
    rows.append((has_back, lay_abs, sub_abs, wnorm, valbad, worst > 1e-4, worst, val, c0))

print('colonnes : arrière / couches absorbantes / substrat absorbant / poids normalisés -> n, valeur ≠ coût, gradient ≠ différences finies')
agg = collections.defaultdict(lambda: [0, 0, 0])
for hb, la, sa, wn, vb, gb, *_ in rows:
    a = agg[(hb, la, sa, wn)]
    a[0] += 1
    a[1] += vb
    a[2] += gb
for k in sorted(agg):
    n, v, g = agg[k]
    print('  arrière=%-5s couches_abs=%-5s sub_abs=%-5s poids_norm=%-5s -> %3d cas | valeur≠coût %3d | gradient≠FD %3d' % (*k, n, v, g))
tot = len(rows)
print('TOTAL %d cas | valeur≠coût %d | gradient≠FD %d' % (tot, sum(r[4] for r in rows), sum(r[5] for r in rows)))
bad = [r for r in rows if r[4]][:3]
for r in bad:
    print('  exemple : val=%.8g coût=%.8g' % (r[7], r[8]), r[:4])
